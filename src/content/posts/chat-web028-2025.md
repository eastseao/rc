## Q1 · 我有几个博客网站页面，想集成到一个APP里，怎么操作（第 1 轮对话）

> **提问：** 我有几个博客网站页面，想集成到一个APP里，怎么操作

非常好 👍，你想把**多个博客网站整合到一个APP**里，有几种不同的做法，取决于你希望的效果（例如是纯浏览展示型，还是能离线阅读、发文、推送通知等）。 我可以一步步帮你实现。 我们先分三类方法：

---

#### 🧩 一、最简单：用 WebView 打包成浏览型 APP（推荐入门方案）

> ✅ 适合你已有多个博客页面，只想通过一个APP来访问和导航。 🕐 几分钟即可生成。

##### 操作流程：

1. 准备好你的网页链接

比如： ```https://seamoonappear.github.io/ https://trtjkqh.github.io/ https://myblog2.github.io/```

1. 创建一个简单导航主页（HTML）

```html
   <!DOCTYPE html>
   <html lang="zh-CN">
   <head>
     <meta charset="UTF-8">
     <meta name="viewport" content="width=device-width, initial-scale=1.0">
     <title>我的博客合集</title>
     <style>
       body { font-family: sans-serif; text-align: center; padding: 30px; }
       a { display: block; margin: 20px; padding: 10px; font-size: 20px; color: #0366d6; text-decoration: none; border: 1px solid #ccc; border-radius: 10px; }
       a:hover { background-color: #f0f0f0; }
     </style>
   </head>
   <body>
     <h1>博客合集导航</h1>
     <a href="https://seamoonappear.github.io/">地平线</a>
     <a href="https://trtjkqh.github.io/">物料查询</a>
     <a href="https://myblog2.github.io/">其他博客</a>
   </body>
   </html>
```

1. 用封装工具生成 APP

有很多**零代码**工具可以直接把网页打包成安卓或 iOS 应用：

| 工具 | 支持系统 | 特点 |
|---|---|---|
| [WebViewGold](https://www.webviewgold.com/) | iOS/Android | 可上传App Store / Google Play |
| [PWABuilder](https://www.pwabuilder.com/) | Android / Windows / Web | 免费且自动生成 |
| [AppGyver](https://www.appgyver.com/) | Android / iOS | 可增加逻辑功能 |
| [Capacitor + Android Studio](https://capacitorjs.com/) | Android / iOS | 稍复杂，开发者方案 |

最推荐初学者： 👉 打开**PWABuilder**→ 输入你的网站导航页网址 → 自动生成安装包。

1. 生成的APK安装到手机即可访问所有博客。

#### 📱 二、进阶：做成「多页面切换」的混合 APP

> ✅ 看起来像原生APP，底部有导航栏、不同标签页对应不同博客。

用框架如：

- Flutter WebView

- React Native + react-native-webview

- Ionic + Capacitor


##### `src/services/db.js`（SQLite 数据层）

```js
// src/services/db.js
import * as SQLite from 'expo-sqlite';

const db = SQLite.openDatabase('blogs.db');

export function initDB() {
  return new Promise((resolve, reject) => {
    db.transaction(tx => {
      // articles: id, guid, title, content_html, summary, link, pubDate, source, cached (0/1)
      tx.executeSql(
        `CREATE TABLE IF NOT EXISTS articles (
          id INTEGER PRIMARY KEY NOT NULL,
          guid TEXT UNIQUE,
          title TEXT,
          content_html TEXT,
          summary TEXT,
          link TEXT,
          pubDate TEXT,
          source TEXT,
          cached INTEGER DEFAULT 0
        )`,
        [],
        () => resolve(),
        (_, err) => { reject(err); return false; }
      );
    });
  });
}

export function insertOrReplaceArticle(article) {
  const { guid, title, content_html, summary, link, pubDate, source } = article;
  return new Promise((resolve, reject) => {
    db.transaction(tx => {
      tx.executeSql(
        `INSERT OR REPLACE INTO articles (guid, title, content_html, summary, link, pubDate, source, cached)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?);`,
        [guid, title, content_html, summary, link, pubDate, source, article.cached ? 1 : 0],
        (_, result) => resolve(result),
        (_, err) => { reject(err); return false; }
      );
    });
  });
}

export function getAllArticles(limit = 200) {
  return new Promise((resolve, reject) => {
    db.transaction(tx => {
      tx.executeSql(
        `SELECT * FROM articles ORDER BY datetime(pubDate) DESC LIMIT ?;`,
        [limit],
        (_, { rows }) => resolve(rows._array),
        (_, err) => { reject(err); return false; }
      );
    });
  });
}

export function searchArticlesByText(q) {
  const pattern = `%${q}%`;
  return new Promise((resolve, reject) => {
    db.transaction(tx => {
      tx.executeSql(
        `SELECT * FROM articles WHERE title LIKE ? OR summary LIKE ? OR content_html LIKE ? ORDER BY datetime(pubDate) DESC;`,
        [pattern, pattern, pattern],
        (_, { rows }) => resolve(rows._array),
        (_, err) => { reject(err); return false; }
      );
    });
  });
}

export function getArticleById(id) {
  return new Promise((resolve, reject) => {
    db.transaction(tx => {
      tx.executeSql(
        `SELECT * FROM articles WHERE id = ?;`,
        [id],
        (_, { rows }) => resolve(rows._array[0]),
        (_, err) => { reject(err); return false; }
      );
    });
  });
}
```

##### `src/services/syncService.js`（抓取 RSS 并存库）

```js
// src/services/syncService.js
import Parser from 'rss-parser';
import axios from 'axios';
import { insertOrReplaceArticle } from './db';

const parser = new Parser({
  timeout: 10000,
});

export async function fetchAndStoreFromRSS(feedUrl, sourceName) {
  try {
    const feed = await parser.parseURL(feedUrl);
    const items = feed.items || [];
    for (const it of items) {
      // Build article object
      const article = {
        guid: it.guid || it.id || it.link,
        title: it.title || '无标题',
        content_html: it['content:encoded'] || it.content || it.summary || '',
        summary: it.summary || '',
        link: it.link || '',
        pubDate: it.isoDate || it.pubDate || (new Date()).toISOString(),
        source: sourceName || feed.title || feedUrl,
        cached: 0
      };
      await insertOrReplaceArticle(article);
    }
    return { success: true, count: items.length };
  } catch (err) {
    console.warn('fetchAndStoreFromRSS error', err.message || err);
    return { success: false, error: err.message || String(err) };
  }
}

// Optional: fetch GitHub Pages posts by listing repo files (if no RSS)
// Example: get raw HTML from a repo path - might need token for private or rate limits
export async function fetchFromGitHubRepoRaw(owner, repo, path = '_posts') {
  const url = `https://api.github.com/repos/${owner}/${repo}/contents/${path}`;
  // If rate-limited, user may provide personal access token
  try {
    const res = await axios.get(url);
    // res.data is array of files; for each markdown/html file you can fetch download_url
    // This is left as an exercise (example provided in docs)
    return res.data;
  } catch (err) {
    throw err;
  }
}
```

> 说明：`rss-parser`的`parseURL`在 React Native 上有时会遇到跨域/https 问题；若有问题可以先用`axios`获取 feed 文本`axios.get(feedUrl).then(r => parser.parseString(r.data))`。

##### `src/services/imageCache.js`（图片缓存示例）

```js
// src/services/imageCache.js
import * as FileSystem from 'expo-file-system';
import { Platform } from 'react-native';
import * as Crypto from 'expo-crypto';

const cacheDir = FileSystem.cacheDirectory + 'images/';

export async function ensureCacheDir() {
  const info = await FileSystem.getInfoAsync(cacheDir);
  if (!info.exists) await FileSystem.makeDirectoryAsync(cacheDir, { intermediates: true });
}

export async function cacheImage(url) {
  try {
    await ensureCacheDir();
    const hash = await Crypto.digestStringAsync(Crypto.CryptoDigestAlgorithm.SHA256, url);
    const ext = url.split('.').pop().split('?')[0];
    const filePath = `${cacheDir}${hash}.${ext}`;
    const info = await FileSystem.getInfoAsync(filePath);
    if (info.exists) return info.uri;
    const dl = await FileSystem.downloadAsync(url, filePath);
    return dl.uri;
  } catch (err) {
    console.warn('cacheImage err', err.message || err);
    return url; // fallback to remote
  }
}
```

##### `src/screens/HomeScreen.js`（主页：列文章 + 搜索 + 同步按钮）

```js
// src/screens/HomeScreen.js
import React, { useEffect, useState } from 'react';
import { View, Text, FlatList, TextInput, Button, TouchableOpacity } from 'react-native';
import { initDB, getAllArticles, searchArticlesByText } from '../services/db';
import ArticleItem from '../components/ArticleItem';
import { fetchAndStoreFromRSS } from '../services/syncService';

const FEEDS = [
  { url: 'https://seamoonappear.github.io/feed.xml', name: '地平线' },
  // Add other RSS feed URLs here
  // { url: 'https://trtjkqh.github.io/feed.xml', name: '物料查询' },
];

export default function HomeScreen({ navigation }) {
  const [articles, setArticles] = useState([]);
  const [q, setQ] = useState('');

  useEffect(() => {
    (async () => {
      await initDB();
      await loadArticles();
    })();
  }, []);

  async function loadArticles() {
    const all = await getAllArticles();
    setArticles(all);
  }

  async function onSearch() {
    if (!q) { await loadArticles(); return; }
    const res = await searchArticlesByText(q);
    setArticles(res);
  }

  async function onSync() {
    // 手动同步：逐个 feed 抓取
    for (const f of FEEDS) {
      await fetchAndStoreFromRSS(f.url, f.name);
    }
    await loadArticles();
  }

  return (
    <View style={{ flex: 1, padding: 12 }}>
      <View style={{ flexDirection: 'row', marginBottom: 8 }}>
        <TextInput
          style={{ flex: 1, borderWidth: 1, borderRadius: 6, padding: 8 }}
          value={q}
          onChangeText={setQ}
          placeholder="搜索标题/正文..."
        />
        <Button title="搜索" onPress={onSearch} />
      </View>

      <View style={{ flexDirection: 'row', justifyContent: 'space-between', marginBottom: 8 }}>
        <Button title="同步 RSS" onPress={onSync} />
        <Button title="刷新列表" onPress={loadArticles} />
      </View>

      <FlatList
        data={articles}
        keyExtractor={(item) => String(item.id)}
        renderItem={({ item }) => (
          <TouchableOpacity onPress={() => navigation.navigate('Article', { id: item.id })}>
            <ArticleItem article={item} />
          </TouchableOpacity>
        )}
      />
    </View>
  );
}
```

```html
// src/screens/ArticleScreen.js
import React, { useEffect, useState } from 'react';
import { View, ScrollView, ActivityIndicator, Button } from 'react-native';

import RenderHtml from 'react-native-render-html';

export default function ArticleScreen({ route }) {

          if (await Sharing.isAvailableAsync()) {
            await Sharing.shareAsync(article.link);
          }
        }} />
      </View>
    </ScrollView>
  );
}
```

# 7) 注意事项与限制

1. RSS 可用性关键 ：该方案对 RSS 依赖高，若博客没有 RSS，需要额外写抓取逻辑或请求作者提供 feed。

1. rss-parser 在 RN 环境 ：少数环境可能需要先用 axios 获取 raw feed，然后用 parser.parseString(xml) 。如果遇到 CORS/https 问题，请告诉我错误日志，我会给解决办法。

1. SQLite FTS ：Expo 的 SQLite 是否支持 FTS5 依赖底层实现；若需要更强检索，请考虑在服务器端创建搜索索引或使用本地 JS 索引库（lunr.js），但 lunr 在移动端性能要评估。

1. 图片与媒体权限 ：在安卓上如果下载很多图片需注意存储配额与权限。

1. GitHub API 限额 ：如使用 GitHub API，可能需要用户 token 来避免被限流。
