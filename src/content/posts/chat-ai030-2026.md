# 🧠 2. embedding.py（支持替换LM Studio）

> 这里先用“简化向量模拟”，后面可以换 LM Studio / sentence-transformers

```python
import hashlib
import numpy as np

def simple_embed(text: str):
    """
    简化 embedding（生产可替换为 LM Studio embedding API）
    """
    h = hashlib.md5(text.encode()).hexdigest()
    vec = np.array([int(h[i:i+2], 16) for i in range(0, 32, 2)])
    return vec / np.linalg.norm(vec)
```

# 🗄️ 3. db.py（记忆存储层）

```python
import sqlite3
import json
from config import DB_PATH

def init_db():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()

    c.execute("""
    CREATE TABLE IF NOT EXISTS memory (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        content TEXT,
        type TEXT,
        embedding TEXT,
        importance REAL,
        tags TEXT,
        timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
        decay REAL DEFAULT 1.0
    )
    """)

    conn.commit()
    conn.close()


def insert_memory(content, mtype, embedding, importance, tags):
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()

    c.execute("""
    INSERT INTO memory (content, type, embedding, importance, tags)
    VALUES (?, ?, ?, ?, ?)
    """, (
        content,
        mtype,
        json.dumps(embedding.tolist()),
        importance,
        json.dumps(tags)
    ))

    conn.commit()
    conn.close()


def fetch_all():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("SELECT * FROM memory")
    rows = c.fetchall()
    conn.close()
    return rows
```

# 🧠 4. cortex.py（核心大脑）

```python
import numpy as np
from db import init_db, insert_memory, fetch_all
from embedding import simple_embed
from config import DECAY_RATE, TOP_K


class CortexMemory:

    def __init__(self):
        init_db()

    # ========== 存储记忆 ==========
    def store(self, content, mtype="fact", importance=0.5, tags=None):
        if tags is None:
            tags = []

        emb = simple_embed(content)

        insert_memory(
            content,
            mtype,
            emb,
            importance,
            tags
        )

    # ========== 语义检索 ==========
    def search(self, query, top_k=TOP_K):
        query_emb = simple_embed(query)
        memories = fetch_all()

        scored = []

        for m in memories:
            emb = np.array(eval(m[3]))
            sim = self.cosine(query_emb, emb)

            # importance + decay 加权
            score = sim * m[4] * m[6]

            scored.append((score, m))

        scored.sort(reverse=True, key=lambda x: x[0])

        return [m[1] for _, m in scored[:top_k]]

    # ========== 学习经验 ==========
    def learn(self, action, context, outcome, insight):
        content = f"""
Action: {action}
Context: {context}
Outcome: {outcome}
Insight: {insight}
"""
        importance = 0.7 if outcome == "success" else 0.4

        self.store(
            content,
            mtype="skill",
            importance=importance,
            tags=["experience"]
        )

    # ========== 遗忘机制 ==========
    def decay(self):
        conn = __import__("sqlite3").connect("cortex_memory.db")
        c = conn.cursor()

        c.execute("""
        UPDATE memory
        SET decay = decay - ?
        """, (DECAY_RATE,))

        conn.commit()
        conn.close()

    # ========== 工具 ==========
    def cosine(self, a, b):
        return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b) + 1e-8)
```
