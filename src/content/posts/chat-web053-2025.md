#### 🧭 整体思路（纯本地运行，无需服务器）

因为浏览器**出于安全原因**，默认不允许网页直接访问本地文件系统（比如`C:/Users/.../data.json`），所以有三种可行方式：

| 方式 | 说明 | 难度 |
|---|---|---|
| **① 用 input 上传 JSON 文件** | 用户点击“导入数据”按钮，选择本地 JSON 文件，JS 用`FileReader`读取内容后展示。 | ✅ 最推荐（零部署） |
| **② 用 VSCode Live Server / Node.js serve 工具** | 在本地开一个静态服务（http://localhost:5500），用`fetch('data/xxx.json')`读取。 | 🔧 稍高级 |
| **③ 用 Electron 或桌面封装** | 把网页打包成桌面 App，直接访问文件系统。 | 🚀 进阶玩法 |

你目前的代码是纯前端 HTML，最适合**方式①：文件上传读取 JSON 并展示**。

#### 🧩 实现步骤（方式①）

我们在你的现有代码中加上实际逻辑（核心是`FileReader`+ 动态生成表格）。

把下面的`<script>`替换到你 index.html 最底部的`<script>`区域 👇：

```html
<script>
/**
 * 文件导入并读取 JSON 内容
 */
function handleFileImport(button, dataType) {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = '.json';
    input.onchange = e => {
        const file = e.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = event => {
            try {
                const jsonData = JSON.parse(event.target.result);
                localStorage.setItem(dataType, JSON.stringify(jsonData)); // 临时保存
                alert(`${dataType} 导入成功，共 ${jsonData.length} 条记录`);
                updateDashboard(); // 更新统计卡片
                renderTable(dataType, jsonData);
            } catch (err) {
                alert('文件格式错误，请检查 JSON 结构');
            }
        };
        reader.readAsText(file);
    };
    input.click();
}

/**
 * 根据导入数据更新仪表盘统计
 */
function updateDashboard() {
    const products = JSON.parse(localStorage.getItem('product') || '[]');
    const boms = JSON.parse(localStorage.getItem('bom') || '[]');
    const purchases = JSON.parse(localStorage.getItem('purchase') || '[]');

    document.querySelector('.stat-card:nth-child(1) .value').textContent = products.length;
    document.querySelector('.stat-card:nth-child(2) .value').textContent = boms.length;
    document.querySelector('.stat-card:nth-child(3) .value').textContent = purchases.length;
}

/**
 * 渲染表格
 */
function renderTable(type, data) {
    if (type === 'product') {
        const tbody = document.querySelector('.table tbody');
        tbody.innerHTML = '';
        data.forEach(item => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td>${item.产品ID || item.id}</td>
                <td>${item.产品名称 || item.name}</td>
                <td><span class="badge bg-success badge-status">${item.状态 || '活跃'}</span></td>
                <td><button class="btn btn-sm btn-outline-primary">查看详情</button></td>
            `;
            tbody.appendChild(tr);
        });
    }
}

/**
 * 初始化绑定按钮
 */
document.addEventListener('DOMContentLoaded', () => {
    document.querySelector('.btn-outline-primary').addEventListener('click', () => handleFileImport(event.target, 'product'));
    document.querySelector('.btn-outline-success').addEventListener('click', () => handleFileImport(event.target, 'bom'));
    document.querySelector('.btn-outline-warning').addEventListener('click', () => handleFileImport(event.target, 'purchase'));
});
</script>
```
