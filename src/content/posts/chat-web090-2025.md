            <input type="text" id="tags" placeholder="输入标签，多个标签用逗号隔开" required>

                <option value="职场">职场</option>

    <script>
        function generateMarkdown() {
            const title = document.getElementById('title').value;
            const tags = document.getElementById('tags').value.split(',').map(tag => tag.trim());
            const category = document.getElementById('category').value;
            const cover = document.getElementById('cover').value;
            const content = document.getElementById('content').value;
            const pinned = document.getElementById('pinned').value;
            const date = new Date().toISOString();
            const lastModified = new Date().toISOString();
            const author = "Gervas";  // 默认作者为 Gervas
            const fileName = title.toLowerCase().replace(/[^a-z0-9]+/g, '-') + '.md'; // 自动生成文件名

            // 生成 Front Matter
            let frontMatter = `---
layout: post
title: "${title}"
subtitle: "文章副标题（可选）"
tags: [${tags.join(', ')}]
category: ${category}
cover: ${cover ? cover : '""'}
pinned: ${pinned}
date: ${date}
last_modified_at: ${lastModified}
author: ${author}
comments: true
mathjax: true
---`;

            // 生成 Markdown 内容
            let markdownContent = frontMatter + `\n\n# ${title}\n\n${content}`;

            // 更新右侧的文件名和 Markdown 内容
            document.getElementById('file-name').value = fileName;
            document.getElementById('markdown-content').textContent = markdownContent;
        }

        // 复制文件名
        function copyFileName() {
            const fileName = document.getElementById('file-name');
            fileName.select();
            document.execCommand('copy');
        }

        // 复制 Markdown 内容
        function copyMarkdown() {
            const markdownContent = document.getElementById('markdown-content');
            const range = document.createRange();
            range.selectNode(markdownContent);
            window.getSelection().removeAllRanges();
            window.getSelection().addRange(range);
            document.execCommand('copy');
        }
    </script>
</body>
</html>

1. 新增内容输入框 ：在表单中添加了一个 <textarea> 输入框，用来输入文章内容。

1. 默认作者为 Gervas ：作者字段已经设置为 "Gervas" ，并且不会被修改。

1. 生成 Markdown 时包含内容 ：生成的 Markdown 内容现在会包括用户输入的内容。
