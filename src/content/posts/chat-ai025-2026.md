import streamlit as st

from skills import market, cost, copywriting

if "messages" not in st.session_state:
    st.session_state.messages = []

# 显示历史消息
for msg in st.session_state.messages:
    with st.chat_message(msg["role"]):
        st.markdown(msg["content"])

# 输入框
user_input = st.chat_input("输入你的需求（市场/成本/文案/产品等）")

if user_input:
    # 显示用户输入
    st.chat_message("user").markdown(user_input)

    # 路由选择技能

    # 调用技能
    if skill_name == "market":
        result = market.run(user_input)

    elif skill_name == "cost":

# 二、核心：桌面窗口 UI（app.py）

直接给你完整可运行版本👇

```python
import tkinter as tk
from tkinter import scrolledtext
from router import route_task
from skills import market, cost, copywriting

# 创建窗口
root = tk.Tk()
root.title("Hermes Agent - 桌面版")
root.geometry("700x500")

# 聊天显示区
chat_area = scrolledtext.ScrolledText(root, wrap=tk.WORD)
chat_area.pack(padx=10, pady=10, fill=tk.BOTH, expand=True)
chat_area.config(state='disabled')

# 输入框
input_box = tk.Entry(root, font=("Arial", 12))
input_box.pack(padx=10, pady=5, fill=tk.X)

def send_message():
    user_input = input_box.get()
    if not user_input:
        return

    # 显示用户输入
    chat_area.config(state='normal')
    chat_area.insert(tk.END, f"\n你：{user_input}\n")

    # 路由技能
    skill_name = route_task(user_input)

    if skill_name == "market":
        result = market.run(user_input)

    elif skill_name == "cost":
        result = cost.run({"cost": 10, "price": 25})

    elif skill_name == "copywriting":
        result = copywriting.run(user_input)

    else:
        result = "未识别需求，请重新输入"

    # 显示结果
    chat_area.insert(tk.END, f"Hermes：{result}\n")
    chat_area.config(state='disabled')

    # 清空输入框
    input_box.delete(0, tk.END)

# 发送按钮
send_btn = tk.Button(root, text="发送", command=send_message)
send_btn.pack(pady=5)

# 回车发送
root.bind('<Return>', lambda event: send_message())

# 启动
root.mainloop()
```

        })

#### 📌 skill_generator.py

```python
import os
import requests

def generate_skill(skill_name, user_goal):
    """
    自动生成新的 Skill 文件
    """

    prompt = f"""
你是一个AI工程师。

请帮我生成一个Python函数Skill：

Skill名称：{skill_name}

功能目标：
{user_goal}

要求：
1. 必须是 def run(input): 格式
2. 返回结构化文本
3. 不要依赖外部库
"""

    res = requests.post(
        "http://localhost:11434/api/generate",
        json={
            "model": "llama3",
            "prompt": prompt,
            "stream": False
        }
    )

    code = res.json()["response"]

    file_path = f"skills/{skill_name}.py"

    with open(file_path, "w", encoding="utf-8") as f:
        f.write(code)

    return file_path
```

SKILL_MAP = {}

def load_skills():
    for file in os.listdir("skills"):
        if file.endswith(".py") and file != "__init__.py":
            name = file.replace(".py", "")
            module = importlib.import_module(f"skills.{name}")

import tkinter as tk
from tkinter import scrolledtext
from planner import plan_task
from executor import execute
from registry import load_skills
from memory import load_memory, save_memory

load_skills()
memory = load_memory()

root = tk.Tk()
root.title("Hermes LM Studio Agent")
root.geometry("950x700")

chat = scrolledtext.ScrolledText(root, font=("Arial", 11))
chat.pack(fill=tk.BOTH, expand=True)

entry = tk.Entry(root, font=("Arial", 12))
entry.pack(fill=tk.X)

def run():
    user_input = entry.get()
    if not user_input:
        return

    chat.insert(tk.END, f"\n你：{user_input}\n")

    # ===== 记忆优化 =====
    if user_input in memory:
        plan = memory[user_input]
        chat.insert(tk.END, "🧠 使用历史最优路径\n")

        plan = plan_task(user_input)

    # ===== 执行 =====
    results = execute(plan)

    for r in results:
        chat.insert(tk.END, f"\n[{r['skill']}]\n{r['result']}\n")

    # ===== 写入记忆 =====
    memory[user_input] = plan
    save_memory(memory)

    entry.delete(0, tk.END)

entry.bind("<Return>", lambda e: run())

tk.Button(root, text="执行", command=run).pack()

root.mainloop()

import importlib
import os
import sys

SKILLS = {}

def get_base_path():
    """兼容 exe 路径"""
    if hasattr(sys, '_MEIPASS'):
        return sys._MEIPASS
    return os.path.abspath(".")

def load_skills():
    """加载所有技能"""
    global SKILLS
    SKILLS.clear()

    base_path = get_base_path()

    # 内置 skills
    skill_paths = [
        os.path.join(base_path, "skills"),
        os.path.join(os.getcwd(), "user_skills")  # 自动生成技能目录
    ]

    for path in skill_paths:
        if not os.path.exists(path):
            continue

        for file in os.listdir(path):
            if file.endswith(".py") and file != "__init__.py":
                name = file.replace(".py", "")

                try:
                    module = importlib.import_module(f"skills.{name}")
                    SKILLS[name] = module.run
                except Exception as e:
                    print(f"加载技能失败: {name} -> {e}")

def get_skill(name):
    """获取技能"""
    return SKILLS.get(name)
