## Q1 · 给我生成一个excel表格转换成json的工具的页面（第 1 轮对话）

> **提问：** 给我生成一个excel表格转换成json的工具的页面

明白，你想要一个可以把 Excel 文件上传并转换成 JSON 的网页工具。我可以帮你生成一个完整的**HTML 页面**，使用**JavaScript + SheetJS (xlsx.js)**来实现这个功能。下面是示例代码：

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Excel 转 JSON 工具</title>
  <script src="https://cdn.jsdelivr.net/npm/xlsx/dist/xlsx.full.min.js"></script>
  <style>
    body {
      font-family: Arial, sans-serif;
      margin: 40px;
      background-color: #f7f7f7;
    }
    h1 {
      text-align: center;
      color: #333;
    }
    .container {
      max-width: 600px;
      margin: auto;
      background: #fff;
      padding: 30px;
      border-radius: 10px;
      box-shadow: 0 0 10px rgba(0,0,0,0.1);
    }
    input[type="file"] {
      display: block;
      margin: 20px 0;
    }
    textarea {
      width: 100%;
      height: 300px;
      margin-top: 20px;
      padding: 10px;
      font-family: monospace;
      font-size: 14px;
    }
    button {
      background-color: #4CAF50;
      color: white;
      padding: 10px 20px;
      border: none;
      cursor: pointer;
      border-radius: 5px;
      font-size: 16px;
    }
    button:hover {
      background-color: #45a049;
    }
  </style>
</head>
<body>

  <div class="container">
    <h1>Excel 转 JSON 工具</h1>
    <input type="file" id="fileInput" accept=".xlsx,.xls">
    <button onclick="convertExcel()">转换成 JSON</button>
    <textarea id="jsonOutput" placeholder="转换结果将在这里显示..." readonly></textarea>
  </div>

  <script>
    function convertExcel() {
      const fileInput = document.getElementById('fileInput');
      const output = document.getElementById('jsonOutput');

      if (!fileInput.files.length) {
        alert('请先选择一个 Excel 文件');
        return;
      }

      const file = fileInput.files[0];
      const reader = new FileReader();

      reader.onload = function(e) {
        const data = new Uint8Array(e.target.result);
        const workbook = XLSX.read(data, { type: 'array' });
        const result = {};

        workbook.SheetNames.forEach(sheetName => {
          const roa = XLSX.utils.sheet_to_json(workbook.Sheets[sheetName], { defval: null });
          if (roa.length) result[sheetName] = roa;
        });

        output.value = JSON.stringify(result, null, 2);
      };

      reader.readAsArrayBuffer(file);
    }
  </script>

</body>
</html>
```

##### 功能说明：

1. 支持上传 `.xlsx`/`.xls`文件 。

1. 使用 SheetJS 解析 Excel。

1. 每个 sheet 都会转换成 JSON 对象，sheet 名称作为 key。

1. 转换结果显示在页面的 textarea 中，可直接复制。

1. 纯前端实现，无需服务器。

  uploadedFiles = Array.from(e.target.files);
  const container = document.getElementById('sheetContainer');
  container.innerHTML = '';
  selectedSheets = {};

  uploadedFiles.forEach((file, idx) => {
    const reader = new FileReader();
    reader.onload = function(e){
      const data = new Uint8Array(e.target.result);

function convertExcel(){
  if(uploadedFiles.length===0){
    alert('请先选择 Excel 文件');
    return;
  }

  const promises = uploadedFiles.map(file => {
    return new Promise(resolve => {
      const reader = new FileReader();
      reader.onload = function(e){
        const data = new Uint8Array(e.target.result);
        const workbook = XLSX.read(data, {type:'array'});
        const result = {};
        (selectedSheets[file.name] || []).forEach(sheetName => {
          const roa = XLSX.utils.sheet_to_json(workbook.Sheets[sheetName], {defval:null});
          if(roa.length) result[sheetName] = roa;
        });
        resolve({fileName:file.name, data:result});
      };
      reader.readAsArrayBuffer(file);
    });
  });

    const merged = {};
    allData.forEach(fileData => {
      Object.keys(fileData.data).forEach(sheet => {
        merged[`${fileData.fileName} - ${sheet}`] = fileData.data[sheet];

}

  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'merged_data.json';
  a.click();
  URL.revokeObjectURL(url);
}
