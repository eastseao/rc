# 🧭 6. Router（任务分发核心）

```python
class Router:

    def route(self, task, agents):
        # 简化版 scoring
        best_agent = None
        best_score = -1

        for agent in agents:
            score = self.score(task, agent)
            if score > best_score:
                best_score = score
                best_agent = agent

        return best_agent

    def score(self, task, agent):
        match = any(c in task.type for c in agent.capabilities)
        return 1 if match else 0
```

# ⚙️ 9. Runtime Engine（核心执行器🔥）

```python
from router import Router
from memory import Memory
from evaluator import Evaluator

class RuntimeEngine:

    def __init__(self, registry):
        self.registry = registry
        self.router = Router()
        self.memory = Memory()
        self.evaluator = Evaluator()

    def execute(self, task):

        # 1. 路由
        agent = self.router.route(task, self.registry.list_agents())

        # 2. 执行
        result = agent.run(task)
        task.result = result
        task.status = "completed"

        # 3. 评分
        task.score = self.evaluator.score(result)

        # 4. 存储
        self.memory.save(task)

        return task
```

class TaskGraph:

    def __init__(self):

        self.tasks = {}

    def add_task(self, task):

        self.tasks[task.id] = task

    def get_ready_tasks(self):

        ready = []

        for task in self.tasks.values():

            if task.status != "pending":
                continue

            if all(
                self.tasks[d].status == "completed"
                for d in task.dependencies
            ):
                ready.append(task)

        return ready
```

class Router:

    def route(self, task, agents):

        best_agent = None

        best_score = -1

        for agent in agents:

            score = self.calculate_score(
                task,
                agent
            )

            if score > best_score:

                best_score = score

                best_agent = agent

        return best_agent

    def calculate_score(
        self,
        task,
        agent
    ):

        score = 0

        if task.context.get(
            "category"
        ) in agent.capabilities:

            score += 1

        score += agent.performance_score

        return score
```

class EventBus:

    def subscribe(
        self,
        event,
        callback
    ):

        self.listeners.setdefault(
            event,
            []
        ).append(callback)

    def publish(
        self,
        event,
        data
    ):

        for callback in self.listeners.get(
            event,
            []

```python
class Scheduler:

    def __init__(
        self,
        graph,
        router,
        registry,
        executor
    ):

        self.graph = graph

        self.router = router

        self.registry = registry

        self.executor = executor

    def run(self):

        while True:

            ready = self.graph.get_ready_tasks()

            if not ready:
                break

            for task in ready:

                agent = self.router.route(
                    task,
                    self.registry.all()
                )

                self.executor.execute(
                    task,
                    agent
                )
```

# 🧠 1. Economy Engine（经济核心）

```python
class EconomyEngine:

    def __init__(self):

        self.companies = []
        self.market = None

    def register_company(self, company):

        self.companies.append(company)

    def run_cycle(self):

        tasks = self.market.generate_tasks()

        allocations = self.allocate_tasks(tasks)

        results = self.execute(allocations)

        self.settle_market(results)

    def allocate_tasks(self, tasks):

        return {
            task: self.select_best_company(task)
            for task in tasks
        }

    def select_best_company(self, task):

        return max(
            self.companies,
            key=lambda c: c.score(task)
        )
```

class CompanyRegistry:

        self.companies = {}

# 🏪 3. Market Engine（市场机制🔥）

```python
class MarketEngine:

    def __init__(self):

        self.tasks = []

    def generate_tasks(self):

        return [
            {
                "type": "content_creation",
                "reward": 10,
                "difficulty": 0.3
            },
            {
                "type": "analysis",
                "reward": 20,
                "difficulty": 0.6
            }
        ]

    def price_task(self, task):

        return task["reward"] * (1 - task["difficulty"])
```

# ⚙️ 5. Resource Allocator（资源分配器）

```python
class ResourceAllocator:

    def allocate(self, companies, resources):

        allocation = {}

        for c in companies:

            allocation[c.name] = resources / len(companies)

        return allocation
```

# 🏆 6. Reputation System（声誉系统🔥）

```python
class ReputationSystem:

    def __init__(self):

        self.scores = {}

    def update(self, company, score):

        self.scores[company] = (
            self.scores.get(company, 0) + score
        )

    def get_rank(self):

        return sorted(
            self.scores.items(),
            key=lambda x: x[1],
            reverse=True
        )
```

# 💰 7. Capital System（资本系统）

```python
class CapitalSystem:

    def __init__(self):

        self.balance = {}

    def reward(self, company, amount):

        self.balance[company] = (
            self.balance.get(company, 0) + amount
        )

    def penalize(self, company, amount):

        self.balance[company] -= amount
```

# ⚔️ 8. Competition Engine（竞争机制🔥）

```python
class CompetitionEngine:

    def compete(self, companies, task):

        scores = {}

        for c in companies:

            scores[c.name] = c.evaluate(task)

        winner = max(scores, key=scores.get)

        return winner
```
