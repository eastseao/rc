function slugify(text) {
  return text.toString().toLowerCase()
    .trim()
    .replace(/[\s]+/g, '-')        // 空格替换为 -
    .replace(/[^\w\-]+/g, '')      // 移除非字母数字和 -
    .replace(/\-\-+/g, '-');       // 多个 - 合并为 1 个
}

function countReadingTime(text) {
  const words = text.trim().split(/\s+/).length;
  const minutes = Math.round(words / 250);
  return minutes || 1;
}

function getCurrentDate() {
  const now = new Date();
  return now.toISOString().split('T')[0];
}

function generateMarkdown() {
  const title = document.getElementById('title').value.trim() || '未命名文章';
  const categories = document.getElementById('categories').value.split(',').map(s => s.trim()).filter(s => s);
  const tags = document.getElementById('tags').value.split(',').map(s => s.trim()).filter(s => s);
  const pin = document.getElementById('pin').value;
  const cover = document.getElementById('cover').value.trim();
  const content = document.getElementById('content').value.trim() || '这里是正文占位符...';

  const date = getCurrentDate();
  const slug = slugify(title);
  const readingTime = countReadingTime(content);

  const fm = [
    '---',
    'layout: post',
    `title: "${title}"`,
    `date: ${date}`,
    `last_modified_at: ${date}`,
    `categories: [${categories.join(', ')}]`,
    `tags: [${tags.join(', ')}]`,
    `pin: ${pin}`,
    cover ? `cover: "${cover}"` : '',
    `阅读时间: 约${readingTime}分钟`,
    '---',
    '',
    content
  ].filter(Boolean).join('\n');

  document.getElementById('output').textContent = fm;
}

function copyMarkdown() {
  const output = document.getElementById('output').textContent;
  navigator.clipboard.writeText(output).then(() => {
    alert('Markdown 已复制到剪贴板！');
  });
}

function downloadMarkdown() {
  const title = document.getElementById('title').value.trim() || '未命名文章';
  const date = getCurrentDate();
  const slug = slugify(title);
  const filename = `_posts/${date}-${slug}.md`;
  const content = document.getElementById('output').textContent;
  const blob = new Blob([content], {type: 'text/markdown;charset=utf-8'});
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}
</script>
</body>
</html>
