##### ✅ 排序逻辑

在`<script>`最下面加入：

```
document.querySelectorAll("#resultTable th").forEach(th => {
  th.addEventListener("click", () => {
    const key = th.dataset.sort;
    if (!key) return;

    sortDirection[key] = !sortDirection[key];

    currentData.sort((a, b) => {
      if (key === "index") return 0; // 序号不参与真实排序
      const v1 = a[key] ?? "";
      const v2 = b[key] ?? "";
      return sortDirection[key] ? (v1 > v2 ? 1 : -1) : (v1 < v2 ? 1 : -1);
    });

    renderTable(currentData);
  });
});
```

根目录/
├─ CP/index.html
├─ cz/data.json
```

如果你的`cz`文件夹就在网站根目录，那么路径正确；如果`cz`在 CP 内部或其他位置，需要调整路径：

```
fetch("/cz/data.json").then(...)
```

或者相对`CP/`：

```
fetch("./../cz/data.json").then(...)
```

---

##### 2️⃣ 类型匹配问题

- BOM 表里每行的物料项目号可能是 数字 ，而 cz/data.json 的 物料项目号 可能是 字符串 。

```
const czItem = czData.find(c => String(c.物料项目号) === String(i.物料项目号));

  [...new Set(a.map(i=>i.可供应渠道).filter(Boolean))].forEach(c=>{
    select.innerHTML+=`<option>${c}</option>`;
  });
  console.log("✅ 数据加载完成", {allData,bomData,czData});
});

function search(){
  const p=document.getElementById("projectInput").value.trim();
  const n=document.getElementById("nameInput").value.trim();
  const c=document.getElementById("channelSelect").value;
  currentData=allData.filter(i=>
    (!p || String(i.项目号)===p) &&
    (!n || i.品名.includes(n)) &&
    (!c || i.可供应渠道===c)
  );
  renderTable();
}

function renderTable(){
  const tbody=document.getElementById("tableBody");
  const table=document.getElementById("resultTable");
  const no=document.getElementById("noResult");

  if(!currentData.length){
    table.classList.add("hidden");
    no.classList.remove("hidden");
    return;
  }

  no.classList.add("hidden");
  table.classList.remove("hidden");

  tbody.innerHTML=currentData.map((i,idx)=>`
    <tr>
      <td>${idx+1}</td>
      <td>${i.项目号}</td>
      <td>${i.品名}</td>
      <td>${i.规格}</td>
      <td>${i.单位}</td>
      <td>${i.零售价||""}</td>
      <td>${i.品牌||""}</td>
      <td>${i.可供应渠道||""}</td>
      <td><button onclick="showBOM('${i.项目号}')" class="text-blue-600 hover:underline">查看BOM</button></td>
    </tr>`).join("");
}

function showBOM(projectId){
  const section=document.getElementById("bomSection");
  const tbody=document.getElementById("bomBody");
  const list=bomData.filter(i=>String(i.项目号)===String(projectId));
  if(!list.length){ section.classList.remove("hidden"); tbody.innerHTML="<tr><td colspan='5' class='text-gray-500'>未找到BOM</td></tr>"; return;}

  tbody.innerHTML=list.map(row=>{
    const cz=czData.find(c=>String(c.物料项目号)===String(row.物料项目号));
    return `
      <tr>
        <td>${row.物料项目号}</td>
        <td>${row.物料名称}</td>
        <td>${row.数量}</td>
        <td>${row.计量单位}</td>
        <td>${cz ? cz.采购单价 : "—"}</td>
      </tr>`;
  }).join("");
  section.classList.remove("hidden");
}

// 点击排序
document.querySelectorAll("th[data-sort]").forEach(th=>{
  th.addEventListener("click",()=>{
    const key=th.dataset.sort;
    if(key==="序号")return;
    sortDirection[key]=!sortDirection[key];
    const dir=sortDirection[key]?1:-1;
    currentData.sort((a,b)=>{
      const x=a[key], y=b[key];
      if(!isNaN(x)&&!isNaN(y)) return (x-y)*dir;
      return String(x??"").localeCompare(String(y??""))*dir;
    });
    document.querySelectorAll("th").forEach(h=>h.classList.remove("sort-asc","sort-desc"));
    th.classList.add(sortDirection[key]?"sort-asc":"sort-desc");
    renderTable();
  });
});
