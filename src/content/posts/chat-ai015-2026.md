```python
def estimate_costs(idea, cost_level):
    """
    根据产品类型 + 成本等级估算成本
    """

    # 基础原料成本模型（可扩展）
    if "鸡胸肉" in idea:
        raw_cost = 6.5
    elif "饮料" in idea:
        raw_cost = 2.5
    elif "坚果" in idea:
        raw_cost = 8.0
    else:
        raw_cost = 5.0

    # 成本等级调整
    level_factor = {
        "低": 0.8,
        "中": 1.0,
        "高": 1.3
    }.get(cost_level, 1.0)

    raw_cost *= level_factor

    # 包装成本
    packaging_cost = 1.2 * level_factor

    # 生产成本
    production_cost = 1.5 * level_factor

    total_cost = raw_cost + packaging_cost + production_cost

    return {
        "raw_material_cost": round(raw_cost, 2),
        "packaging_cost": round(packaging_cost, 2),
        "production_cost": round(production_cost, 2),
        "total_cost": round(total_cost, 2)
    }


def calculate_profit(cost_data, channel):
    """
    根据渠道计算售价和利润
    """

    total_cost = cost_data["total_cost"]

    # 渠道扣点
    channel_fee_rate = {
        "电商": 0.25,
        "商超": 0.35,
        "代理": 0.4
    }.get(channel, 0.25)

    # 目标毛利率
    target_margin = 0.5

    # 反推售价
    suggested_price = total_cost / (1 - channel_fee_rate - target_margin)

    # 实际利润
    profit = suggested_price * (1 - channel_fee_rate) - total_cost

    return {
        "suggested_price": round(suggested_price, 2),
        "channel_fee_rate": channel_fee_rate,
        "profit_per_unit": round(profit, 2),
        "profit_margin": round(profit / suggested_price, 2)
    }
```
