```html
    function searchData() {
```

下面我给你改好**核心表格生成部分**的逻辑，保持原来的查询、红色渐变风格，同时实现合并展示：

```html
// 构建合并行的表格
function buildMergedTable(filtered) {
  if (filtered.length === 0) return "<p class='text-center text-gray-500'>未找到匹配的数据。</p>";

  // 先按产品字段排序
  filtered.sort((a, b) => {
    const keyA = `${a["产品项目号"]}|${a["产品名称"]}|${a["规格"]}|${a["品牌"]}|${a["系列"]}`;
    const keyB = `${b["产品项目号"]}|${b["产品名称"]}|${b["规格"]}|${b["品牌"]}|${b["系列"]}`;
    return keyA.localeCompare(keyB);
  });

  let html = `<table class="w-full text-sm text-center border border-gray-200 rounded-lg shadow-sm">
    <thead class="bg-red-600 text-white">
      <tr>
        <th class="p-2">产品项目号</th>
        <th class="p-2">产品名称</th>
        <th class="p-2">规格</th>
        <th class="p-2">品牌</th>
        <th class="p-2">系列</th>
        <th class="p-2">物料项目号</th>
        <th class="p-2">物料名称</th>
        <th class="p-2">数量</th>
      </tr>
    </thead>
    <tbody>`;

  let i = 0;
  while (i < filtered.length) {
    const current = filtered[i];
    const key = `${current["产品项目号"]}|${current["产品名称"]}|${current["规格"]}|${current["品牌"]}|${current["系列"]}`;

    // 找出重复的数量
    let rowspan = 1;
    for (let j = i + 1; j < filtered.length; j++) {
      const next = filtered[j];
      const nextKey = `${next["产品项目号"]}|${next["产品名称"]}|${next["规格"]}|${next["品牌"]}|${next["系列"]}`;
      if (nextKey === key) rowspan++;
      else break;
    }

    // 输出合并行
    for (let k = 0; k < rowspan; k++) {
      const row = filtered[i + k];
      html += `<tr class="${(i+k) % 2 === 0 ? 'bg-gray-50' : 'bg-white'} hover:bg-red-50 transition">`;
      if (k === 0) {
        html += `<td class="p-2" rowspan="${rowspan}">${row["产品项目号"] || ""}</td>`;
        html += `<td class="p-2" rowspan="${rowspan}">${row["产品名称"] || ""}</td>`;
        html += `<td class="p-2" rowspan="${rowspan}">${row["规格"] || ""}</td>`;
        html += `<td class="p-2" rowspan="${rowspan}">${row["品牌"] || ""}</td>`;
        html += `<td class="p-2" rowspan="${rowspan}">${row["系列"] || ""}</td>`;
      }
      html += `<td class="p-2">${row["物料项目号"] || ""}</td>`;
      html += `<td class="p-2">${row["物料名称"] || ""}</td>`;
      html += `<td class="p-2">${row["数量"] || ""}</td>`;
      html += `</tr>`;
    }
    i += rowspan;
  }

  html += `</tbody></table>`;
  return html;
}
```

#### ✅ 一、常见原因分析

| 原因 | 说明 | 解决方法 |
|---|---|---|
| ①`data.json`路径错误 | `fetch('data.json')`默认找同一目录 | 确保`data.json`和`index.html`在同一文件夹（`bom/`）下 |
| ② 文件名或大小写错误 | GitHub Pages 区分大小写 | 确认文件名为`data.json`，不是`Data.json` |
| ③ JSON 格式错误 | 多了逗号、缺少方括号等 | 可用[https://jsonlint.com](https://jsonlint.com)验证格式 |
| ④ GitHub 缓存未刷新 | 新上传的文件还未生效 | 等几分钟或刷新浏览器缓存（Ctrl + F5） |
| ⑤ CORS 或路径问题 | 用绝对路径`/bom/data.json`会导致 404 | 必须使用相对路径`fetch('data.json')` |

###### **情况 1：JavaScript 没加载或放错位置**

你可能把`<script>`标签放在`<head>`，但`searchData()`依赖的元素还没加载完。 👉**解决方法**：把脚本放在`</body>`前。

```html

<script>
  async function searchData() {
    const query = document.getElementById('search').value.trim();
    const tableBody = document.getElementById('tableBody');
    tableBody.innerHTML = "<tr><td colspan='8' class='text-center text-gray-400'>⏳ 数据加载中...</td></tr>";

    try {
      const response = await fetch('data.json');
      const data = await response.json();
      const results = data.filter(item =>
        Object.values(item).some(v => String(v).includes(query))
      );

      if (results.length === 0) {
        tableBody.innerHTML = "<tr><td colspan='8' class='text-center text-gray-400'>未找到匹配项</td></tr>";
        return;

      // 显示数据
      tableBody.innerHTML = results.map(r => `

          <td>${r.产品名称}</td>
          <td>${r.规格}</td>
          <td>${r.品牌}</td>
          <td>${r.系列}</td>
          <td>${r.物料项目号}</td>
          <td>${r.物料名称}</td>
          <td>${r.数量}</td>
        </tr>
      `).join('');
    } catch (e) {
```

###### **情况 2：`onclick="searchData()"`写在按钮上时，函数作用域不对**

```html
<button id="searchBtn">查询</button>
```
