现在页面放在根目录/，你要把它移到kanban/index.html，那地址就变成：

https://trtjkqh.github.io/kanban/

一个小坑来了：你的所有路径都是绝对路径/xxx，一旦放到/kanban/下，就会 404，因为/assets/img/trtlogo.png会指向根目录的 assets，而不是/kanban/assets。

你需要把代码里的所有以/开头的路径改成相对路径，例如：

原路径改成
/assets/img/trtlogo.png../assets/img/trtlogo.png
/CP/index.html../CP/index.html
/bom/index.html../bom/index.html
/tz/index.html../tz/index.html
/supplier/index.html../supplier/index.html

也就是说，把你的按钮部分这样改：

<button class="nav-btn ..." data-url="../CP/index.html" data-tooltip="成品查询">
...
<button class="nav-btn ..." data-url="../bom/index.html" data-tooltip="产品BOM">
...
<button class="nav-btn ..." data-url="../tz/index.html" data-tooltip="物料查询">
...
<button class="nav-btn ..." data-url="../supplier/index.html" data-tooltip="供应商查询">
...

Logo 路径也改：

<img src="../assets/img/trtlogo.png" alt="Logo">

如果未来你可能继续搬目录、想偷懒，建议把资源用**相对路径./**开头，比如：

<img src="./assets/img/trtlogo.png">

然后在kanban目录单独放assets，结构就更稳定。

一句话总结：只要你把页面放进子目录，绝对路径/xxx就会失效，要改成../xxx或./xxx。

你现在放到/kanban/，改上面那几处就能正常跑了。

<script>
  const correctPassword = "666888"; 
  const overlay = document.getElementById('passwordOverlay');
  const input = document.getElementById('passwordInput');
  const errorText = document.getElementById('passwordError');
  const box = document.getElementById('passwordBox');

  if (sessionStorage.getItem("authPassed") === "true") overlay.style.display = "none";

  function checkPassword() {
    const val = input.value.trim();
    if (val === correctPassword) {
      sessionStorage.setItem("authPassed","true");
      overlay.style.opacity = "0";
      setTimeout(() => overlay.style.display = "none", 300);
    } else {
      errorText.classList.remove("hidden");
      box.classList.add("shake");
      input.classList.add("border-red-500","ring-red-400");
      setTimeout(()=>{ box.classList.remove("shake"); input.classList.remove("ring-red-400"); },400);
    }
  }
  input.addEventListener("keydown", e => { if (e.key === "Enter") checkPassword(); });

  const buttons = document.querySelectorAll('.nav-btn');
  const contentArea = document.getElementById('contentArea');
  const tooltip = document.getElementById('tooltip');
  const sidebar = document.getElementById('sidebar');
  const toggleBtn = document.getElementById('toggleBtn');
  const logo = document.getElementById('logoContainer');

  buttons.forEach(btn => {
    btn.addEventListener('click', () => {
      const url = btn.getAttribute('data-url');
      contentArea.innerHTML = `<iframe src="${url}"></iframe>`;
    });

    btn.addEventListener('mouseenter', () => {
      if (btn.querySelector('.btn-text').style.display === 'none') {
        tooltip.textContent = btn.getAttribute('data-tooltip');
        tooltip.style.opacity = '1';
        const rect = btn.getBoundingClientRect();
        tooltip.style.top = rect.top + rect.height / 2 - tooltip.offsetHeight / 2 + 'px';
        tooltip.style.left = rect.right + 8 + 'px';
      }
    });
    btn.addEventListener('mouseleave', () => tooltip.style.opacity = '0');
  });

  function collapseSidebar() {
    sidebar.classList.remove('w-64'); sidebar.classList.add('w-16');
    logo.querySelector('img').style.display='none';
    document.querySelectorAll('.btn-text').forEach(el => el.style.display='none');
    document.querySelectorAll('.btn-icon').forEach(el => el.style.display='inline');
  }
  function expandSidebar() {
    sidebar.classList.remove('w-16'); sidebar.classList.add('w-64');
    logo.querySelector('img').style.display='';
    document.querySelectorAll('.btn-text').forEach(el => el.style.display='');
    document.querySelectorAll('.btn-icon').forEach(el => el.style.display='none');
  }

  collapseSidebar();
  toggleBtn.addEventListener('click', () => {
    sidebar.classList.contains('w-64') ? collapseSidebar() : expandSidebar();
  });
</script>
