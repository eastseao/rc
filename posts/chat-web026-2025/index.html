      const supplier = document.getElementById('supplier').value.trim();

Q2 · ❌ 无法读取数据文件，请检查路径 tz/data.json（第 2 轮对话）

提问： ❌ 无法读取数据文件，请检查路径 tz/data.json

非常好，我们来一步步定位问题。这个错误说明页面没能正确加载tz/data.json文件。

下面是最常见的5 种原因与对应解决办法👇

✅ 一、路径问题（最常见）

检查你的文件目录结构。 确保是👇这样摆放的：

📁 你的项目根目录
│
├─ index.html
└─ 📁 tz
   └─ data.json

👉注意事项：

  • 文件夹名字必须是 tz （大小写一致）。

  • data.json 必须放在 tz 文件夹里。

  • 你的 index.html 必须和 tz 文件夹在 同一级目录 。

如果不是这样，请调整。

✅ 二、在本地直接用浏览器打开（file://）时

浏览器的安全策略（CORS）会禁止读取本地 JSON 文件。症状：用 Chrome 打开会报

“Access to fetch at ‘file:///…/tz/data.json’ from origin ‘null’ has been blocked by CORS policy”

🔧 解决方案有三种：
  1. 最简单 ：用 VSCode 的插件 “Live Server” 启动网页
  • 右键index.html→ 选择「Open with Live Server」 - 页面地址变成http://127.0.0.1:5500/index.html- 此时能正常加载 JSON。
  1. 或者 ：用 Python 启一个本地服务器
   python -m http.server 8000

然后浏览器打开： http://localhost:8000/index.html

  1. 或者 ：放到 GitHub Pages

例如仓库结构： your-repo/ ├─ index.html └─ tz/data.json推送后在https://你的用户名.github.io/your-repo/ 打开就能用了。

✅ 三、文件名或扩展名不正确

请确认文件名是：

data.json

不是：

data.json.txt

可以右键查看文件属性或在 VSCode 中看右上角文件类型。

        console.log("✅ 数据已加载:", dataCache.length, "条");
      })
      .catch(err => {
        console.error("❌ 数据加载失败:", err);
        alert("无法读取数据文件，请确认路径 data.json 是否正确。");
      });

    function searchData() {
      const supplier = document.getElementById("supplier").value.trim();
      const projectNo = document.getElementById("projectNo").value.trim();
      const projectName = document.getElementById("projectName").value.trim();

      if (!supplier && !projectNo && !projectName) {
        alert("请输入至少一项查询条件！");
        return;
      }

      let filtered = dataCache.filter(item => {
        const supplierStr = item["供应商名称"] ? String(item["供应商名称"]) : "";
        const projectNoStr = item["项目号"] ? String(item["项目号"]) : "";
        const projectNameStr = item["项目名称"] ? String(item["项目名称"]) : "";

        const matchSupplier = supplier ? supplierStr.includes(supplier) : true;
        const matchProjectNo = projectNo ? projectNoStr.includes(projectNo) : true;
        const matchProjectName = projectName ? projectNameStr.includes(projectName) : true;

        return matchSupplier && matchProjectNo && matchProjectName;
      });

      // 按合同编号从最新到最旧排序
      filtered.sort((a, b) => {
        const aNum = Number(a["合同编号"].match(/\d+/g).join(''));
        const bNum = Number(b["合同编号"].match(/\d+/g).join(''));
        return bNum - aNum;
      });

      if (filtered.length === 0) {
        document.getElementById("result").innerHTML = "<p>未找到匹配的数据。</p>";
        return;
      }

      let total = filtered.reduce((sum, item) => sum + Number(item["订单总额"] || 0), 0);

      let tableHTML = `
        <table>
          <thead>
            <tr>
              <th>合同编号</th>
              <th>供应商名称</th>
              <th>项目号</th>
              <th>项目名称</th>
              <th>数量</th>
              <th>单位</th>
              <th>采购单价</th>
              <th>订单总额</th>
            </tr>
          </thead>
          <tbody>
            ${filtered.map(row => `
              <tr>
                <td>${row["合同编号"] || ""}</td>
                <td>${row["供应商名称"] || ""}</td>
                <td>${row["项目号"] || ""}</td>
                <td>${row["项目名称"] || ""}</td>
                <td>${row["数量"] || ""}</td>
                <td>${row["单位"] || ""}</td>
                <td>${row["采购单价"] || ""}</td>
                <td>${row["订单总额"] || ""}</td>
              </tr>
            `).join('')}
          </tbody>
          <tfoot>
            <tr><td colspan="7">金额总计</td><td>${total.toFixed(2)}</td></tr>
          </tfoot>
        </table>
      `;

      document.getElementById("result").innerHTML = tableHTML;
    }
  </script>
