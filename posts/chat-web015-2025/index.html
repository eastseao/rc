提问： chirpy markdown博文生成器 那最终的功能确认如下： 单文件前端生成器 双栏布局：左侧输入、右侧输出 Front Matter 固定 layout: post 自动生成 date、last_modified_at 支持分类、标签（自定义） 支持置顶（pin: true/false） 支持封面图片、视频、音乐 自动计算阅读时间：每 250 字/分钟 → 阅读时间: 约X分钟（四舍五入） 输出 Markdown 文件名 _posts/YYYY-MM-DD-slug.md 复制、下载功能 输出框带正文占位符 无实时渲染 我现在可以生成 最终 v2.2 单文件 HTML

明白了，你最终想做的是Chirpy Markdown 博文生成器 v2.2，单文件 HTML 前端工具，功能确认如下，我帮你整理成最终规格说明，确保无遗漏：

功能列表 v2.2（最终版）

1️⃣ 基本布局

  • 单文件前端生成器，完全前端运行，无后端依赖。

  • 双栏布局：

  • 左侧：用户输入栏 - 右侧：输出 Markdown 预览/生成框（带占位符正文）。

  • 输出不实时渲染（即用户输入不会自动生成 HTML 预览，只输出 Markdown 文件）。

2️⃣ Front Matter 固定字段

  • layout: post （固定）

  • date → 自动生成当前日期

  • last_modified_at → 自动生成当前日期

  • categories → 用户自定义

  • tags → 用户自定义

  • pin → true / false（置顶）

  • cover → 图片 / 视频 / 音乐（用户上传或填 URL）

3️⃣ 阅读时间计算

  • 规则：每 250 字/分钟

  • 自动计算阅读时间

  • 输出格式： 阅读时间: 约X分钟 ，四舍五入

4️⃣ 文件生成

  • 输出 Markdown 文件名： _posts/YYYY-MM-DD-slug.md （slug 可根据标题生成，空格替换为 - ）

  • 输出框带正文占位符：用户只需输入正文即可

5️⃣ 功能增强

  • 复制 Markdown 到剪贴板

  • 下载 生成的 Markdown 文件

<script>
function slugify(text) {
  return text.toString().toLowerCase()
    .trim()
    .replace(/[\s]+/g, '-')        // 空格替换为 -
    .replace(/[^\w\-]+/g, '')      // 移除非字母数字和 -
    .replace(/\-\-+/g, '-');       // 多个 - 合并为 1 个
}

function countReadingTime(text) {
  const words = text.trim().split(/\s+/).length;
  const minutes = Math.round(words / 250);
  return minutes || 1;
}

function getCurrentDate() {
  const now = new Date();
  return now.toISOString().split('T')[0];
}

function generateMarkdown() {
  const title = document.getElementById('title').value.trim() || '未命名文章';
  const categories = document.getElementById('categories').value.split(',').map(s => s.trim()).filter(s => s);
  const tags = document.getElementById('tags').value.split(',').map(s => s.trim()).filter(s => s);
  const pin = document.getElementById('pin').value;
  const cover = document.getElementById('cover').value.trim();
  const content = document.getElementById('content').value.trim() || '这里是正文占位符...';

  const date = getCurrentDate();
  const slug = slugify(title);
  const readingTime = countReadingTime(content);

  const fm = [
    '---',
    'layout: post',
    `title: "${title}"`,
    `date: ${date}`,
    `last_modified_at: ${date}`,
    `categories: [${categories.join(', ')}]`,
    `tags: [${tags.join(', ')}]`,
    `pin: ${pin}`,
    cover ? `cover: "${cover}"` : '',
    `阅读时间: 约${readingTime}分钟`,
    '---',
    '',
    content
  ].filter(Boolean).join('\n');

  document.getElementById('output').textContent = fm;
}

function copyMarkdown() {
  const output = document.getElementById('output').textContent;
  navigator.clipboard.writeText(output).then(() => {
    alert('Markdown 已复制到剪贴板！');
  });
}

function downloadMarkdown() {
  const title = document.getElementById('title').value.trim() || '未命名文章';
  const date = getCurrentDate();
  const slug = slugify(title);
  const filename = `_posts/${date}-${slug}.md`;
  const content = document.getElementById('output').textContent;
  const blob = new Blob([content], {type: 'text/markdown;charset=utf-8'});
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}
</script>
</body>
</html>

function slugify(text) {
  return text.toString().toLowerCase()
    .trim()
    .replace(/[\s]+/g, '-')        
    .replace(/[^\w\-]+/g, '')      
    .replace(/\-\-+/g, '-');       
}

function countReadingTime(text) {
  const words = text.trim().split(/\s+/).length;
  const minutes = Math.round(words / 250);
  return minutes || 1;
}

function getCurrentDate() {
  const now = new Date();
  return now.toISOString().split('T')[0];
}

function generateMarkdown() {
  const leftTitle = document.getElementById('title').value.trim() || '未命名文章';
  const categories = document.getElementById('categories').value.split(',').map(s => s.trim()).filter(s => s);
  const tags = document.getElementById('tags').value.split(',').map(s => s.trim()).filter(s => s);
  const pin = document.getElementById('pin').value;
  const cover = document.getElementById('cover').value.trim();
  const content = document.getElementById('content').value.trim() || '这里是正文占位符...';

  // 同步左侧标题到右侧生成标题输入框
  const generatedTitleInput = document.getElementById('generated-title');
  if (!generatedTitleInput.value) generatedTitleInput.value = leftTitle;

  const title = generatedTitleInput.value.trim();
  const date = getCurrentDate();
  const slug = slugify(title);
  const readingTime = countReadingTime(content);

  const fm = [
    '---',
    'layout: post',
    `title: "${title}"`,
    `date: ${date}`,
    `last_modified_at: ${date}`,
    `categories: [${categories.join(', ')}]`,
    `tags: [${tags.join(', ')}]`,
    `pin: ${pin}`,
    cover ? `cover: "${cover}"` : '',
    `阅读时间: 约${readingTime}分钟`,
    '---',
    '',
    content
  ].filter(Boolean).join('\n');

  document.getElementById('output').textContent = fm;
}

function copyMarkdown() {
  const output = document.getElementById('output').textContent;
  navigator.clipboard.writeText(output).then(() => {
    alert('Markdown 已复制到剪贴板！');
  });
}

function downloadMarkdown() {
  const title = document.getElementById('generated-title').value.trim() || '未命名文章';
  const date = getCurrentDate();
  const slug = slugify(title);
  const filename = `_posts/${date}-${slug}.md`;
  const content = document.getElementById('output').textContent;
  const blob = new Blob([content], {type: 'text/markdown;charset=utf-8'});
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}

function copyTitle() {
  const title = document.getElementById('generated-title').value.trim() || '未命名文章';
  navigator.clipboard.writeText(title).then(() => {
    alert('标题已复制到剪贴板！');
  });
}

function nowFormatted(){
  const d = new Date();
  const pad = n => String(n).padStart(2,'0');
  const YYYY = d.getFullYear();
  const MM = pad(d.getMonth()+1);
  const DD = pad(d.getDate());
  const hh = pad(d.getHours());
  const mm = pad(d.getMinutes());
  const ss = pad(d.getSeconds());
  return `${YYYY}-${MM}-${DD} ${hh}:${mm}:${ss} +0800`;
}
