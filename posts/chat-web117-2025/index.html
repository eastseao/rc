Q2 · 导出的表格，在名称最前面加入项目号（第 2 轮对话）

提问： 导出的表格，在名称最前面加入项目号

明白了，你希望在导出的 Excel 文件名中，把项目号放在前面，如果有项目号就用“项目号-产品名称.xlsx”，如果没有项目号就用“产品名称.xlsx”。

你可以把导出按钮的事件稍微修改一下，如下：

// 导出 Excel
el('exportButton').addEventListener('click', () => {
  const data = [
    ['项目号', inputs.project.value],
    ['产品名称', inputs.product.value],
    ['制费系数 (%)', inputs.r.value],
    ['原料（含税）', inputs.raw.value],
    ['包材（含税）', inputs.pack.value],
    ['其他运费', inputs.freight.value],
    ['制造费用合计', outputs.manu.value],
    ['市场定价毛利率 (%)', inputs.margin.value],
    ['折扣 (%)', inputs.disc.value],
    ['合计成本（不含税）', outputs.cost.textContent],
    ['出厂价', outputs.price.textContent],
    ['市场零售价', outputs.retail.textContent],
    ['实际毛利率', outputs.actual.textContent]
  ];

  const ws = XLSX.utils.aoa_to_sheet(data);
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, '成本核算');

  // 文件名：项目号-产品名称.xlsx
  let filename = '';
  if (inputs.project.value) {
    filename += inputs.project.value + '-';
  }
  filename += inputs.product.value || '成本核算';
  filename += '.xlsx';

  XLSX.writeFile(wb, filename);
});

这样导出时，如果你输入了项目号“P123”，产品名称“牛奶”，导出的文件名就会是：

P123-牛奶.xlsx

没有项目号就只显示产品名称或默认“成本核算.xlsx”。

<script>
  const el = id => document.getElementById(id);
  const inputs = {
    project: el('projectNumber'),
    product: el('productName'),
    r: el('overheadRate'),
    raw: el('rawMaterial'),
    pack: el('packaging'),
    freight: el('otherFreight'),
    margin: el('profitMargin'),
    disc: el('discount')
  };
  const outputs = {
    manu: el('manufacturingCost'),
    cost: el('totalCost'),
    price: el('factoryPrice'),
    retail: el('retailPrice'),
    actual: el('actualMargin'),
    error: el('errorMessage')
  };

  function getNum(input) {
    const v = parseFloat(input.value);
    return isNaN(v) ? 0 : v;
  }

  function recalculate() {
    const r = getNum(inputs.r) / 100;
    const raw = getNum(inputs.raw);
    const pack = getNum(inputs.pack);
    const freight = getNum(inputs.freight);
    const margin = getNum(inputs.margin) / 100;
    const discPercent = getNum(inputs.disc);

    outputs.error.classList.add('hidden');

    const numerator = (raw + pack) / (1 + r) + freight;
    const denominator = (1 - margin) / (1 + r) - r;

    if (Math.abs(denominator) < 1e-8 || denominator <= 0) {
      outputs.error.textContent = '参数无效：分母 ≤ 0，无法计算出厂价';
      outputs.error.classList.remove('hidden');
      return;
    }

    const factoryPrice = numerator / denominator;
    const manufacturingCost = factoryPrice * r;
    const totalCost = (raw + pack) / (1 + r) + freight + manufacturingCost;

    let retailPrice = discPercent > 0 ? factoryPrice / (discPercent / 100) : Infinity;

    let actualMargin = factoryPrice !== 0 ? ((factoryPrice / (1 + r) - totalCost) / factoryPrice) * (1 + r) : 0;

    outputs.manu.value = manufacturingCost.toFixed(2);
    outputs.cost.textContent = totalCost.toFixed(2);
    outputs.price.textContent = factoryPrice.toFixed(2);
    outputs.retail.textContent = isFinite(retailPrice) ? retailPrice.toFixed(2) : '—';
    outputs.actual.textContent = (actualMargin * 100).toFixed(2) + '%';
  }

  Object.values(inputs).forEach(inp => inp.addEventListener('input', recalculate));
  recalculate();

  // 导出 Excel
  el('exportButton').addEventListener('click', () => {
    const data = [
      ['项目号', inputs.project.value],
      ['产品名称', inputs.product.value],
      ['制费系数 (%)', inputs.r.value],
      ['原料（含税）', inputs.raw.value],
      ['包材（含税）', inputs.pack.value],
      ['其他运费', inputs.freight.value],
      ['制造费用合计', outputs.manu.value],
      ['市场定价毛利率 (%)', inputs.margin.value],
      ['折扣 (%)', inputs.disc.value],
      ['合计成本（不含税）', outputs.cost.textContent],
      ['出厂价', outputs.price.textContent],
      ['市场零售价', outputs.retail.textContent],
      ['实际毛利率', outputs.actual.textContent]
    ];

    const ws = XLSX.utils.aoa_to_sheet(data);

    // 表格美化：首行加粗居中
    const range = XLSX.utils.decode_range(ws['!ref']);
    for (let C = range.s.c; C <= range.e.c; ++C) {
      const cell = ws[XLSX.utils.encode_cell({r:0, c:C})];
      if (cell) cell.s = { font: { bold: true }, alignment: { horizontal: "center" } };
    }

    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, '成本核算');

    let filename = '';
    if (inputs.project.value) filename += inputs.project.value + '-';
    filename += inputs.product.value || '成本核算';
    filename += '.xlsx';

    XLSX.writeFile(wb, filename);
  });
</script>

<script>
  const el = id => document.getElementById(id);
  const inputs = {
    project: el('projectNumber'),
    product: el('productName'),
    r: el('overheadRate'),
    raw: el('rawMaterial'),
    pack: el('packaging'),
    freight: el('otherFreight'),
    margin: el('profitMargin'),
    disc: el('discount')
  };
  const outputs = {
    manu: el('manufacturingCost'),
    cost: el('totalCost'),
    price: el('factoryPrice'),
    retail: el('retailPrice'),
    actual: el('actualMargin'),
    error: el('errorMessage')
  };

  function getNum(input) {
    const v = parseFloat(input.value);
    return isNaN(v) ? 0 : v;
  }

  function recalculate() {
    const r = getNum(inputs.r) / 100;
    const raw = getNum(inputs.raw);
    const pack = getNum(inputs.pack);
    const freight = getNum(inputs.freight);
    const margin = getNum(inputs.margin) / 100;
    const discPercent = getNum(inputs.disc);

    outputs.error.classList.add('hidden');

    const numerator = (raw + pack) / (1 + r) + freight;
    const denominator = (1 - margin) / (1 + r) - r;

    if (Math.abs(denominator) < 1e-8 || denominator <= 0) {
      outputs.error.textContent = '参数无效：分母 ≤ 0，无法计算出厂价';
      outputs.error.classList.remove('hidden');
      return;
    }

    const factoryPrice = numerator / denominator;
    const manufacturingCost = factoryPrice * r;
    const totalCost = (raw + pack) / (1 + r) + freight + manufacturingCost;

    let retailPrice = discPercent > 0 ? factoryPrice / (discPercent / 100) : Infinity;
    let actualMargin = factoryPrice !== 0 ? ((factoryPrice / (1 + r) - totalCost) / factoryPrice) * (1 + r) : 0;

    outputs.manu.value = manufacturingCost.toFixed(2);
    outputs.cost.textContent = totalCost.toFixed(2);
    outputs.price.textContent = factoryPrice.toFixed(2);
    outputs.retail.textContent = isFinite(retailPrice) ? retailPrice.toFixed(2) : '—';
    outputs.actual.textContent = (actualMargin * 100).toFixed(2) + '%';
  }

  Object.values(inputs).forEach(inp => inp.addEventListener('input', recalculate));
  recalculate();

  el('exportButton').addEventListener('click', () => {
    const data = [
      ['项目号', inputs.project.value],
      ['产品名称', inputs.product.value],
      ['制费系数 (%)', inputs.r.value],
      ['原料（含税）', inputs.raw.value],
      ['包材（含税）', inputs.pack.value],
      ['其他运费', inputs.freight.value],
      ['制造费用合计', outputs.manu.value],
      ['市场定价毛利率 (%)', inputs.margin.value],
      ['折扣 (%)', inputs.disc.value],
      ['合计成本（不含税）', outputs.cost.textContent],
      ['出厂价', outputs.price.textContent],
      ['市场零售价', outputs.retail.textContent],
      ['实际毛利率', outputs.actual.textContent]
    ];

    const ws = XLSX.utils.aoa_to_sheet(data);

    // 美化首行
    const range = XLSX.utils.decode_range(ws['!ref']);
    for (let C = range.s.c; C <= range.e.c; ++C) {
      const cell = ws[XLSX.utils.encode_cell({r:0, c:C})];
      if (cell) cell.s = { font: { bold: true }, alignment: { horizontal: "center" } };
    }

    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, '成本核算');

    // 文件名：项目号-产品名称-成本核算.xlsx
    let filename = '';
    if (inputs.project.value) filename += inputs.project.value + '-';
    filename += inputs.product.value || '成本核算';
    filename += '-成本核算.xlsx';

    XLSX.writeFile(wb, filename);
  });
</script>
