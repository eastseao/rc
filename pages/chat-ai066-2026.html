# 🧭 6. Router（任务分发核心）

class Router:

    def route(self, task, agents):
        # 简化版 scoring
        best_agent = None
        best_score = -1

        for agent in agents:
            score = self.score(task, agent)
            if score > best_score:
                best_score = score
                best_agent = agent

        return best_agent

    def score(self, task, agent):
        match = any(c in task.type for c in agent.capabilities)
        return 1 if match else 0

from router import Router
from memory import Memory

class RuntimeEngine:

    def __init__(self, registry):
        self.registry = registry
        self.router = Router()
        self.memory = Memory()
        self.evaluator = Evaluator()

    def execute(self, task):

        # 1. 路由
        agent = self.router.route(task, self.registry.list_agents())

        # 2. 执行
        result = agent.run(task)
        task.result = result
        task.status = "completed"

        # 3. 评分
        task.score = self.evaluator.score(result)

        # 4. 存储
        self.memory.save(task)

        return task

# 2️⃣ workflow/task_graph.py

DAG 核心。

class TaskGraph:

    def __init__(self):

        self.tasks = {}

    def add_task(self, task):

        self.tasks[task.id] = task

    def get_ready_tasks(self):

        ready = []

        for task in self.tasks.values():

            if task.status != "pending":
                continue

            if all(
                self.tasks[d].status == "completed"
                for d in task.dependencies
            ):
                ready.append(task)

        return ready

# 4️⃣ core/router.py

能力路由器。

class Router:

    def route(self, task, agents):

        best_agent = None

        best_score = -1

        for agent in agents:

            score = self.calculate_score(
                task,
                agent
            )

            if score > best_score:

                best_score = score

                best_agent = agent

        return best_agent

    def calculate_score(
        self,
        task,
        agent
    ):

        score = 0

        if task.context.get(
            "category"
        ) in agent.capabilities:

            score += 1

        score += agent.performance_score

        return score

class Executor:

    def execute(
        self,
        task,
        agent
    ):

        try:

            result = agent.run(task)

            task.result = result

            task.status = "completed"

        except Exception as e:

            task.status = "failed"

            task.result = {
                "error": str(e)

        return task

class EventBus:

    def __init__(self):

        self.listeners = {}

    def subscribe(
        self,
        event,
        callback
    ):

        self.listeners.setdefault(
            event,
            []
        ).append(callback)

    def publish(
        self,
        event,
        data
    ):

        for callback in self.listeners.get(
            event,
            []
        ):

            callback(data)

# 9️⃣ core/scheduler.py

调度器。

class Scheduler:

    def __init__(
        self,
        graph,
        router,
        registry,
        executor
    ):

        self.graph = graph

        self.router = router

        self.registry = registry

        self.executor = executor

    def run(self):

        while True:

            ready = self.graph.get_ready_tasks()

            if not ready:
                break

            for task in ready:

                agent = self.router.route(
                    task,
                    self.registry.all()
                )

                self.executor.execute(
                    task,
                    agent
                )

from openai import AsyncOpenAI

class OpenAIProvider:

    def __init__(
        self,
        api_key
    ):
        self.client = AsyncOpenAI(
            api_key=api_key
        )

    async def generate(
        self,
        prompt,
        system=None
    ):

        response = await self.client.chat.completions.create(
            model="gpt-5",
            messages=[
                {
                    "role":"system",
                    "content":system
                },
                {
                    "role":"user",
                    "content":prompt
                }
            ]
        )

        return response.choices[0].message.content

# 🧠 5. Vector Store（向量记忆层）

class VectorStore:

    def __init__(self):

        self.vectors = []

    def add(self, embedding, metadata):

        self.vectors.append(
            {
                "embedding":embedding,

# 🧠 1. Economy Engine（经济核心）

class EconomyEngine:

    def __init__(self):

        self.companies = []
        self.market = None

    def register_company(self, company):

        self.companies.append(company)

    def run_cycle(self):

        tasks = self.market.generate_tasks()

        allocations = self.allocate_tasks(tasks)

        results = self.execute(allocations)

        self.settle_market(results)

    def allocate_tasks(self, tasks):

        return {
            task: self.select_best_company(task)
            for task in tasks
        }

    def select_best_company(self, task):

        return max(
            self.companies,
            key=lambda c: c.score(task)
        )
