function buildMergedTable(filtered) {
  if (filtered.length === 0) return "<p class='text-center text-gray-500'>未找到匹配的数据。</p>";

  // 先按产品字段排序
  filtered.sort((a, b) => {
    const keyA = `${a["产品项目号"]}|${a["产品名称"]}|${a["规格"]}|${a["品牌"]}|${a["系列"]}`;
    const keyB = `${b["产品项目号"]}|${b["产品名称"]}|${b["规格"]}|${b["品牌"]}|${b["系列"]}`;
    return keyA.localeCompare(keyB);

    // 找出重复的数量
    let rowspan = 1;
    for (let j = i + 1; j < filtered.length; j++) {

      const nextKey = `${next["产品项目号"]}|${next["产品名称"]}|${next["规格"]}|${next["品牌"]}|${next["系列"]}`;
      if (nextKey === key) rowspan++;
      else break;
    }

    // 输出合并行
    for (let k = 0; k < rowspan; k++) {
      const row = filtered[i + k];
      html += `<tr class="${(i+k) % 2 === 0 ? 'bg-gray-50' : 'bg-white'} hover:bg-red-50 transition">`;
      if (k === 0) {
        html += `<td class="p-2" rowspan="${rowspan}">${row["产品项目号"] || ""}</td>`;

        html += `<td class="p-2" rowspan="${rowspan}">${row["品牌"] || ""}</td>`;

      html += `<td class="p-2">${row["物料名称"] || ""}</td>`;
      html += `<td class="p-2">${row["数量"] || ""}</td>`;
      html += `</tr>`;
    }
    i += rowspan;
  }

  html += `</tbody></table>`;
  return html;
