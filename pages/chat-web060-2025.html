async function showPage(page) {
  const content = document.getElementById('content');
  const data = await fetch(`data/${page}.json`).then(res => res.json());
  content.innerHTML = generateTable(data);
}

function generateTable(data) {
  if (!data.length) return '<p>暂无数据</p>';
  const keys = Object.keys(data[0]);
  const header = keys.map(k => `<th>${k}</th>`).join('');
  const rows = data.map(d => `<tr>${keys.map(k => `<td>${d[k]}</td>`).join('')}</tr>`).join('');
  return `<table><thead><tr>${header}</tr></thead><tbody>${rows}</tbody></table>`;
}

  const headers = tableWrap.querySelectorAll('th.sortable');

      const key = th.dataset.key;

        if (a[key] == null) return 1;

      saveModuleData(currentModule, currentData);

  });
}

// js/render.js

export function getColumnsFromData(data) {
  if (!data || data.length === 0) return [];
  const keys = new Set();
  data.forEach(row => Object.keys(row || {}).forEach(k => {
    if (k === '_id') return;
    keys.add(k);
  }));
  return Array.from(keys);
}

export function renderTable(data, { onEdit, onDelete } = {}) {
  const columns = getColumnsFromData(data);
  // header
  let thead = '<tr>';
  columns.forEach(col => {
    thead += `<th class="p-2 text-left sortable cursor-pointer" data-key="${col}" data-dir="asc">${col}</th>`;
  });
  thead += '<th class="p-2 text-left">操作</th></tr>';

  // rows
  let rows = data.map(row => {
    const tds = columns.map(col => {
      const val = row[col] == null ? '' : row[col];
      return `<td class="p-2 align-top">${escapeHtml(String(val))}</td>`;
    }).join('');
    const id = row._id || '';
    return `<tr class="border-t"><td colspan="100%" style="display:none"></td></tr><tr>${tds}<td class="p-2 align-top">
      <button data-id="${id}" class="edit-btn text-sm px-2 py-1 mr-2 bg-white border rounded">编辑</button>
      <button data-id="${id}" class="del-btn text-sm px-2 py-1 bg-red-50 text-red-600 border rounded">删除</button>
    </td></tr>`;
  }).join('');

  const table = `
    <table class="min-w-full table-fixed">
      <thead class="text-sm text-gray-600">${thead}</thead>
      <tbody class="text-sm text-gray-700">${rows}</tbody>
    </table>
  `;
  // after render attach events
  setTimeout(() => {
    document.querySelectorAll('.edit-btn').forEach(b => b.onclick = (e) => {
      const id = e.currentTarget.dataset.id;
      onEdit && onEdit(id);
    });
    document.querySelectorAll('.del-btn').forEach(b => b.onclick = (e) => {
      const id = e.currentTarget.dataset.id;
      onDelete && onDelete(id);
    });
  }, 0);

  return table;
}

export function renderFormFields(container, fields, values = {}) {
  // fields: array of keys
  fields.forEach(k => {
    const val = values[k] ?? '';
    const label = `<label class="block text-sm text-gray-600 mb-1">${k}</label>`;
    const inputHtml = `<input name="${k}" type="text" value="${escapeHtml(val)}" class="w-full px-3 py-2 border rounded" />`;
    container.insertAdjacentHTML('beforeend', `<div>${label}${inputHtml}</div>`);
  });
}

function escapeHtml(unsafe) {
  return unsafe.replace(/[&<"'>]/g, function(m) {
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[m];
  });
}

// js/storage.js

export async function loadModuleData(module) {
  const local = localStorage.getItem(module);
  if (local) {
    try {
      return JSON.parse(local);
    } catch(e) {
      console.warn('local parse error', e);
    }
  }
  // fetch initial JSON from data/
  try {
    const res = await fetch(`data/${module}.json`);
    if (!res.ok) return [];
    const data = await res.json();
    // ensure each record has _id
    const withId = data.map(d => ({ ...d, _id: d._id || Date.now() + Math.floor(Math.random()*10000) }));
    // do not overwrite localStorage; only return
    return withId;
  } catch (e) {
    console.error(e);
    return [];
  }
}

/** Download JSON file */
export function downloadJSON(module, data) {

  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `${module}_${new Date().toISOString().slice(0,10)}.json`;
  a.click();
  URL.revokeObjectURL(url);
}

/** Export to Excel using SheetJS (XLSX global) */
export function exportToExcel(module, data) {
  if (!window.XLSX) {
    alert('XLSX 库未加载，无法导出。');
    return;
  }
  const ws = XLSX.utils.json_to_sheet(data.map(d => {
    const copy = { ...d };
    delete copy._id;
    return copy;
  }));
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, module);
  XLSX.writeFile(wb, `${module}_${new Date().toISOString().slice(0,10)}.xlsx`);
}

  fetch('data/bom.json').then(r=>r.json()).then(d=>{bomData=d; renderTable('bom',bomData);});

function renderTable(module,data){
  let tbody=document.getElementById(module+'Table').querySelector('tbody');
  tbody.innerHTML='';
  data.forEach(row=>{
    let tr=document.createElement('tr');
    for(let key in row){
      let td=document.createElement('td');
      td.contentEditable=true;
      td.textContent=row[key];
      tr.appendChild(td);
    }
    tbody.appendChild(tr);
  });
}

function addRow(module){
  let data=[];
  if(module==='supplier') data=supplierData;
  else if(module==='bom') data=bomData;
  let keys=Object.keys(data[0] || {});
  let newRow={};
  keys.forEach(k=>newRow[k]='');
  data.push(newRow);
  renderTable(module,data);
}

function saveData(module){
  let tbody=document.getElementById(module+'Table').querySelector('tbody');
  let rows=tbody.querySelectorAll('tr');
  let data=[];
  rows.forEach(tr=>{
    let obj={};
    tr.querySelectorAll('td').forEach((td,i)=>{
      let key=Object.keys((module==='supplier'?supplierData[0]:bomData[0]))[i];
      obj[key]=td.textContent;
    });
    data.push(obj);
  });
  if(module==='supplier') supplierData=data;
  else if(module==='bom') bomData=data;
  const blob=new Blob([JSON.stringify(data,null,2)],{type:'application/json'});
  const url=URL.createObjectURL(blob);
  const a=document.createElement('a');
  a.href=url;
  a.download=module+'.json';
  a.click();
  URL.revokeObjectURL(url);
}

function exportExcel(module){
  let data=(module==='supplier'?supplierData:bomData);
  let ws=XLSX.utils.json_to_sheet(data);
  let wb=XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb,ws,module);
  XLSX.writeFile(wb,module+'.xlsx');
}

function importFile(event,module){
  const file=event.target.files[0];
  if(!file) return;
  const reader=new FileReader();
  const fileName=file.name.toLowerCase();
  if(fileName.endsWith('.json')){
    reader.onload=()=>importJSON(module,reader.result);
    reader.readAsText(file);
  }else if(fileName.endsWith('.xlsx')){
    reader.onload=(evt)=>importExcel(module,evt.target.result);
    reader.readAsBinaryString(file);
  }else alert('仅支持 JSON 或 Excel');
}

function importJSON(module,content){
  try{
    const data=JSON.parse(content);
    if(module==='supplier') supplierData=data;
    else if(module==='bom') bomData=data;
    renderTable(module,data);
    alert(`成功导入 ${data.length} 条记录`);
  }catch{
    alert('JSON格式错误');
  }
}

function importExcel(module,binaryData){
  const workbook=XLSX.read(binaryData,{type:'binary'});
  const sheetName=workbook.SheetNames[0];
  const data=XLSX.utils.sheet_to_json(workbook.Sheets[sheetName]);
  if(module==='supplier') supplierData=data;
  else if(module==='bom') bomData=data;
  renderTable(module,data);
  alert(`成功从 Excel 导入 ${data.length} 条记录`);
}

      }]
    },
    options:{responsive:true,plugins:{legend:{display:false}}}
  });
}

function importModuleFile(evt, module) {
  const file = evt.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  const name = file.name.toLowerCase();
  if(name.endsWith('.json')) {
    reader.onload = () => {
      try {
        const data = JSON.parse(reader.result);
        setModuleData(module, data, { saveLocal:true });
        alert(`成功导入 ${file.name} (${data.length} 条)`);
      } catch(e) { alert('JSON 解析错误'); }
    };
    reader.readAsText(file);
  } else if(name.endsWith('.xlsx') || name.endsWith('.xls')) {
    reader.onload = (e) => {
      try {
        const wb = XLSX.read(e.target.result, { type: 'binary' });
        const sheet = wb.SheetNames[0];
        const data = XLSX.utils.sheet_to_json(wb.Sheets[sheet]);
        setModuleData(module, data, { saveLocal:true });
        alert(`成功从 Excel 导入 ${file.name} (${data.length} 条)`);
      } catch(err) { alert('Excel 解析错误'); }
    };
    reader.readAsBinaryString(file);
  } else {
    alert('仅支持 .json 或 .xlsx/.xls 文件');
  }
}

function importModuleFile(evt, module) {
  const file = evt.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  const name = file.name.toLowerCase();
  if(name.endsWith('.json')) {
    reader.onload = () => {
      try {
        const data = JSON.parse(reader.result);
        setModuleData(module, data, { saveLocal:true });
        alert(`成功导入 ${file.name}（${data.length} 条）`);
      } catch(e) { alert('JSON 解析错误：' + e.message); }
    };
    reader.readAsText(file);
  } else if(name.endsWith('.xlsx') || name.endsWith('.xls')) {
    reader.onload = (e) => {
      try {
        const wb = XLSX.read(e.target.result, { type: 'binary' });
        const sheet = wb.SheetNames[0];
        const data = XLSX.utils.sheet_to_json(wb.Sheets[sheet]);
        setModuleData(module, data, { saveLocal:true });
        alert(`成功从 Excel 导入 ${file.name}（${data.length} 条）`);
      } catch(err) { alert('Excel 解析错误：' + err.message); }
    };
    reader.readAsBinaryString(file);
  } else {
    alert('仅支持 .json 或 .xlsx/.xls 文件');
  }
}

/* 合并重复：按五个字段合并数量 */
function mergeBOMDuplicates(){
  if(!bomData || bomData.length===0){ alert('BOM 数据为空'); return; }
  // 使用当前显示的表格中的数据作为合并对象 if table visible, else use bomData
  const table = document.getElementById('bomTable');
  const tbody = table.querySelector('tbody');
  const rows = Array.from(tbody.querySelectorAll('tr'));
  // gather current displayed data (if hidden, prompt)
  if(rows.length===0){ alert('请先查询以显示数据，然后使用合并功能。'); return; }
  // build array of objects from table
  const displayed = rows.map(tr=>{
    const tds = Array.from(tr.querySelectorAll('td'));
    // header names in order:
    const keys = ['项目号','品名','规格','零售价（元）','品牌','物料项目号','物料名称','数量','计量单位'];
    const obj = {};
    keys.forEach((k,i)=> obj[k] = (tds[i] ? tds[i].textContent.trim() : ''));
    return obj;
  });
  // group by key fields
  const groups = {};
  displayed.forEach(item=>{
    const key = [item['项目号'], item['品名'], item['规格'], item['零售价（元）'], item['品牌']].map(s=> (s||'').trim()).join('||');
    if(!groups[key]) groups[key] = { items: [], sample: item };
    groups[key].items.push(item);
  });
  // prepare merged result
  let merged = [];
  let mergedCount = 0, mergedGroups = 0;
  Object.values(groups).forEach(g=>{
    if(g.items.length===1){
      merged.push(g.sample);
    } else {
      // merge: sum 数量 (best-effort numeric)
      let sum = 0;
      g.items.forEach(it=> {
        const n = Number((it['数量']||'').toString().replace(/,/g,'')) || 0;
        sum += n;
      });
      const first = Object.assign({}, g.sample);
      first['数量'] = sum;
      merged.push(first);
      mergedGroups += 1;
      mergedCount += g.items.length;
    }
  });
  // replace displayed records back into bomData: careful—we will replace only those records that match any of displayed keys
  // Strategy: compute keys set for displayed; remove from bomData those matching keys; then unshift merged items to top (so they appear)
  const displayedKeys = new Set(displayed.map(item => [item['项目号'], item['品名'], item['规格'], item['零售价（元）'], item['品牌']].map(s=> (s||'').trim()).join('||')));
  // remove matching from bomData
  bomData = bomData.filter(item => {
    const key = [item['项目号'], item['品名'], item['规格'], item['零售价'] || item['零售价（元）'], item['品牌']].map(s=> (s||'').toString().trim()).join('||');
    return !displayedKeys.has(key);
  });
  // add merged items to top
  merged.reverse().forEach(it => bomData.unshift(it));
  // persist to localStorage
  localStorage.setItem(STORAGE_KEYS.bom, JSON.stringify(bomData));
  // re-query to reflect result (using current query inputs)
  queryBOM();
  document.getElementById('merge-result').textContent = `合并完成：${mergedGroups} 组，涉及 ${mergedCount} 条记录，已写入并更新表格。`;
}

function mergeBOMDuplicates(){
  if(!bomData || bomData.length===0){ alert('BOM 数据为空'); return; }
  const table = document.getElementById('bomTable');
  const rows = Array.from(table.querySelectorAll('tbody tr'));
  if(rows.length===0){ alert('请先查询以显示数据，然后使用合并功能。'); return; }
  const displayed = rows.map(tr=>{
    const tds = Array.from(tr.querySelectorAll('td'));
    const keys = ['项目号','品名','规格','零售价（元）','品牌','物料项目号','物料名称','数量','计量单位'];
    const obj = {}; keys.forEach((k,i)=> obj[k] = (tds[i] ? tds[i].textContent.trim() : '')); return obj;
  });
  const groups = {};
  displayed.forEach(item=>{
    const key = [item['项目号'], item['品名'], item['规格'], item['零售价（元）'], item['品牌']].map(s=>(s||'').trim()).join('||');
    if(!groups[key]) groups[key] = { items: [], sample: item };
    groups[key].items.push(item);
  });
  const merged = [];
  let mergedGroups=0, mergedCount=0;
  Object.values(groups).forEach(g=>{
    if(g.items.length===1) merged.push(g.sample);
    else {
      let sum=0; g.items.forEach(it=> { const n=Number((it['数量']||'').toString().replace(/,/g,''))||0; sum+=n; });
      const first = Object.assign({}, g.sample); first['数量'] = sum; merged.push(first);
      mergedGroups++; mergedCount += g.items.length;
    }
  });
  // Remove displayed keys from bomData and insert merged to top
  const displayedKeys = new Set(displayed.map(item => [item['项目号'], item['品名'], item['规格'], item['零售价（元）'], item['品牌']].map(s=> (s||'').trim()).join('||')));
  bomData = bomData.filter(item => {
    const key = [item['项目号'], item['品名'], item['规格'], item['零售价'] || item['零售价（元）'], item['品牌']].map(s=> (s||'').toString().trim()).join('||');
    return !displayedKeys.has(key);
  });
  merged.reverse().forEach(it => bomData.unshift(it));
  localStorage.setItem(STORAGE_KEYS.bom, JSON.stringify(bomData));
  queryBOM();
  document.getElementById('merge-result').textContent = `合并完成：${mergedGroups} 组，涉及 ${mergedCount} 条，已更新。`;
}

function mergeBOMDuplicates(){
  const table=document.getElementById('bomTable');
  const rows=Array.from(table.querySelectorAll('tbody tr'));
  if(rows.length===0){ alert('请先查询出要合并的 BOM 记录'); return; }
  const displayed = rows.map(tr=> {
    const tds=Array.from(tr.querySelectorAll('td'));
    const keys=['项目号','品名','规格','零售价（元）','品牌','物料项目号','物料名称','数量','计量单位'];
    const obj={}; keys.forEach((k,i)=> obj[k]= (tds[i]?tds[i].textContent.trim():'') ); return obj;
  });
  const groups={};
  displayed.forEach(it=>{
    const key=[it['项目号'],it['品名'],it['规格'],it['零售价（元）'],it['品牌']].map(s=>(s||'').trim()).join('||');
    if(!groups[key]) groups[key]=[]; groups[key].push(it);
  });
  const mergedList=[];
  let groupsCount=0, itemsCount=0;
  Object.values(groups).forEach(arr=>{
    if(arr.length===1) mergedList.push(arr[0]);
    else {
      groupsCount++; itemsCount += arr.length;
      let sum=0; arr.forEach(a=> sum += Number(a['数量']||0) || 0);
      const keep = Object.assign({}, arr[0]); keep['数量'] = sum; mergedList.push(keep);
    }
  });
  // Remove displayed keys from original bomData and insert merged items to top
  const displayedKeys = new Set(displayed.map(it=> [it['项目号'],it['品名'],it['规格'],it['零售价（元）'],it['品牌']].map(s=> (s||'').trim()).join('||')));
  bomData = bomData.filter(item => {
    const key = [item['项目号'], item['品名'], item['规格'], item['零售价'] || item['零售价（元）'], item['品牌']].map(s=> (s||'').toString().trim()).join('||');
    return !displayedKeys.has(key);
  });
  mergedList.reverse().forEach(m=> bomData.unshift(m));
  localStorage.setItem(STORAGE_KEYS.bom, JSON.stringify(bomData));
  queryBOM();
  document.getElementById('merge-result').textContent = `合并完成：${groupsCount} 组，涉及 ${itemsCount} 条。`;
}
