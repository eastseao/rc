✅ Step 2：.github/workflows/pages.yml

.github/workflows/pages.yml

内容如下👇

name: Build and Deploy Jekyll site

on:
  push:
    branches:
      - main  # 你的默认分支
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout source
        uses: actions/checkout@v4

      - name: Setup Ruby
        uses: ruby/setup-ruby@v1
        with:
          ruby-version: '3.1'
          bundler-cache: true

      - name: Install dependencies
        run: bundle install

      - name: Build site with Jekyll
        run: bundle exec jekyll build

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v3
        with:
          path: ./_site

  deploy:
    runs-on: ubuntu-latest
    needs: build
    permissions:
      pages: write
      id-token: write

    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}

    steps:
      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4

function nowFormatted(){
  const d = new Date();
  const pad = n => String(n).padStart(2,'0');
  const YYYY = d.getFullYear();
  const MM = pad(d.getMonth()+1);
  const DD = pad(d.getDate());
  const hh = pad(d.getHours());
  const mm = pad(d.getMinutes());
  const ss = pad(d.getSeconds());
  return `${YYYY}-${MM}-${DD} ${hh}:${mm}:${ss} +0800`;
}

function slugify(text){
  if(!text) return 'post';
  let s = String(text).trim().toLowerCase();
  s = s.replace(/\s+/g,'-').replace(/[^\w\-]/g,'-').replace(/\-+/g,'-').replace(/^-+|-+$/g,'');
  return s || 'post';
}

---

🔹 一、侧边栏导航不显示的原因

  1. 页面 Front Matter 不完整或不规范

- 必须有 layout: page 才会被 Chirpy 识别为侧边栏页面 - 必须有 title（显示在导航上） - order 可选，但推荐指定，控制顺序 - 如果缺少这些字段，页面不会出现在侧边栏

  1. 文件位置与 URL 不匹配

- _tabs/ 下的文件会自动生成 URL，例如 _tabs/about.md → /about/ - navbar 中的 URL 必须与生成的 URL 一致 - 如果 navbar 的 URL 拼写错误或加了多余的斜杠，点击会无效

  1. 根目录冲突

- 如果根目录同时存在 index.md 和 index.html，GitHub Pages 构建可能只选择一个 - 导致页面模板或侧边栏初始化失败

  1. 主题 CSS/JS 没加载

- 如果主题文件未正确加载，侧边栏 DOM 无法渲染 - 可能是 _config.yml 或 Gemfile 配置不完整
