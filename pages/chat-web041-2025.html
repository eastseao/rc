# 5. 持久化与导出（你提到“导出 Excel”） 业务需求回顾：你之前提过“将四个板块数据导出到同一个 Excel 文件的一个页面里”。我给出两种简单实现：

方案 A — 兼容性好：导出为 CSV（单文件，包含板块列）

优点：无需第三方库，兼容性佳；缺点：不是多 sheet。

示例代码（合并四个板块到一个 CSV，并触发下载）：

function exportToCSV(allPanelsData /* Array of objects with panelName, items */) {
  // allPanelsData = [{ panel: '板块A', items: [{title, date, priority, tags}, ...] }, ...]
  const rows = [];
  // header
  rows.push(['板块','标题','日期','优先级','标签','备注'].join(','));

  allPanelsData.forEach(panel => {
    panel.items.forEach(item => {
      const line = [
        `"${panel.panel}"`,
        `"${(item.title||'').replace(/"/g,'""')}"`,
        `"${item.date||''}"`,
        `"${item.priority||''}"`,
        `"${(item.tags||[]).join(';')}"`,
        `"${(item.note||'').replace(/"/g,'""')}"`
      ].join(',');
      rows.push(line);
    });
  });

  const csvContent = rows.join('\n');
  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `每日待办_${new Date().toISOString().slice(0,10)}.csv`;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
}

方案 B — 多 Sheet 的真实 Excel（推荐：使用 SheetJS）

优点：每个板块一个 sheet，格式和 Excel 兼容好；缺点：需要引入第三方库（SheetJS xlsx），但常用且可靠。

示例（需要在页面加入 <script src="https://cdn.jsdelivr.net/npm/xlsx/dist/xlsx.full.min.js"></script>）：

<!-- 在你的 HTML 底部引入 SheetJS（只在可信环境下使用 CDN） -->
<script src="https://cdn.jsdelivr.net/npm/xlsx/dist/xlsx.full.min.js"></script>
function exportToXLSX(allPanelsData) {
  // allPanelsData = [{ panel: '板块A', items: [{title, date, priority, tags}, ...] }, ...]
  const wb = XLSX.utils.book_new();

  allPanelsData.forEach(panel => {
    // convert items to sheet rows
    const data = panel.items.map(it => ({
      标题: it.title || '',
      日期: it.date || '',
      优先级: it.priority || '',
      标签: (it.tags||[]).join(';'),
      备注: it.note || ''
    }));
    const ws = XLSX.utils.json_to_sheet(data);
    // sheet name length <=31
    const sheetName = panel.panel.slice(0, 31);
    XLSX.utils.book_append_sheet(wb, ws, sheetName);
  });

  const filename = `每日待办_${new Date().toISOString().slice(0,10)}.xlsx`;
  XLSX.writeFile(wb, filename);
}

提醒：如果你担心 CDN 问题，可把 xlsx.full.min.js 打包到你项目里或使用私有静态托管。

# 6. 颜色设置相关：实现、持久化、应用逻辑（带代码） 我把“实时预览 + 保存到 localStorage + 应用到任务卡”三件事写成一套 JS：

// --- 常量 key ---
const COLOR_SETTINGS_KEY = 'todo_color_settings_v1';

// 默认值
const DEFAULT_COLOR_SETTINGS = {
  highBg: '#ff0000', highText: '#ffffff',
  mediumBg: '#ffff00', mediumText: '#000000',
  lowBg: '#e5e7eb', lowText: '#000000'
};

// 载入 & 应用到 CSS 变量
function loadColorSettings() {
  const saved = localStorage.getItem(COLOR_SETTINGS_KEY);
  const settings = saved ? JSON.parse(saved) : DEFAULT_COLOR_SETTINGS;
  applyColorsToCSSVars(settings);
  // 更新 inputs & previews
  document.getElementById('highBgColor').value = settings.highBg;
  document.getElementById('highTextColor').value = settings.highText;
  document.getElementById('mediumBgColor').value = settings.mediumBg;
  document.getElementById('mediumTextColor').value = settings.mediumText;
  document.getElementById('lowBgColor').value = settings.lowBg;
  document.getElementById('lowTextColor').value = settings.lowText;
  updatePreviewBoxes(settings);
  return settings;
}

function applyColorsToCSSVars(s) {
  const root = document.documentElement;
  root.style.setProperty('--priority-high-bg', s.highBg);
  root.style.setProperty('--priority-high-text', s.highText);
  root.style.setProperty('--priority-medium-bg', s.mediumBg);
  root.style.setProperty('--priority-medium-text', s.mediumText);
  root.style.setProperty('--priority-low-bg', s.lowBg);
  root.style.setProperty('--priority-low-text', s.lowText);
}

// 更新 preview 小方块
function updatePreviewBoxes(s) {
  document.getElementById('highPreview').style.background = s.highBg;
  document.getElementById('highPreview').style.color = s.highText;
  document.getElementById('mediumPreview').style.background = s.mediumBg;
  document.getElementById('mediumPreview').style.color = s.mediumText;
  document.getElementById('lowPreview').style.background = s.lowBg;
  document.getElementById('lowPreview').style.color = s.lowText;
}

// 读取表单并保存
function applyColorSettings() {
  const settings = {
    highBg: document.getElementById('highBgColor').value,
    highText: document.getElementById('highTextColor').value,
    mediumBg: document.getElementById('mediumBgColor').value,
    mediumText: document.getElementById('mediumTextColor').value,
    lowBg: document.getElementById('lowBgColor').value,
    lowText: document.getElementById('lowTextColor').value
  };
  applyColorsToCSSVars(settings);
  updatePreviewBoxes(settings);
  localStorage.setItem(COLOR_SETTINGS_KEY, JSON.stringify(settings));
  showToast('颜色设置已保存');
  // 也可以触发任务卡重渲染以确保所有卡更新样式
}

// 还原默认
function resetColorSettings() {
  localStorage.removeItem(COLOR_SETTINGS_KEY);
  applyColorsToCSSVars(DEFAULT_COLOR_SETTINGS);
  updatePreviewBoxes(DEFAULT_COLOR_SETTINGS);
  showToast('颜色已重置为默认');
}

// 实时预览：绑定 input change 事件
['highBgColor','highTextColor','mediumBgColor','mediumTextColor','lowBgColor','lowTextColor'].forEach(id=>{
  const el = document.getElementById(id);
  if(el) el.addEventListener('input', () => {
    const tmp = {
      highBg: document.getElementById('highBgColor').value,
      highText: document.getElementById('highTextColor').value,
      mediumBg: document.getElementById('mediumBgColor').value,
      mediumText: document.getElementById('mediumTextColor').value,
      lowBg: document.getElementById('lowBgColor').value,
      lowText: document.getElementById('lowTextColor').value
    };
    applyColorsToCSSVars(tmp);
    updatePreviewBoxes(tmp);
  });
});

// 页面初始化时调用
document.addEventListener('DOMContentLoaded', () => {
  loadColorSettings();
  document.getElementById('applyColorBtn').addEventListener('click', applyColorSettings);
  document.getElementById('resetColorBtn').addEventListener('click', resetColorSettings);
});

// 简单 toast
function showToast(msg) {
  // 你可以用已有的 toast 组件，这里给个简易实现
  const el = document.createElement('div');
  el.textContent = msg;
  el.setAttribute('role', 'status');
  el.className = 'fixed bottom-4 right-4 bg-black text-white p-2 rounded shadow';
  document.body.appendChild(el);
  setTimeout(()=> el.remove(), 2500);
}

# 8. 性能 要点

  • 当任务数量很大（>500）时，使用“虚拟化列表”技术（比如 Virtualized List）或分页，避免一次性 DOM 渲染过多。
  • 防抖搜索（debounce）。
  • 减少重绘：颜色改动只改 CSS 变量，避免遍历所有卡做 style 改写。
  • 如果使用第三方库（SheetJS, Sortable），按需加载（动态 import）以减小首屏包体。

# 9. 安全

  • 对用户输入做 XSS 防护：任何 HTML 显示要转义或使用安全的富文本编辑器。

  • 导出/导入功能做输入检查，避免解析恶意文件。

// 格式化日期为 MM-DD
function formatDate(dateStr) {
  const d = new Date(dateStr);
  const mm = String(d.getMonth()+1).padStart(2,'0');
  const dd = String(d.getDate()).padStart(2,'0');
  return `${mm}-${dd}`;
}

// 修改 updateTask 保存日期为 MM-DD
function updateTask(index,key,value){
  if(key==='startDate' || key==='endDate'){
    const d = new Date(value);
    value = `${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`;
  }
  tasks[index][key]=value;
  saveTasks();
}
