<script>
  const correctPassword = "666888"; // 你可以改成自己的

  function checkPassword() {
    const input = document.getElementById('passwordInput').value;
    const error = document.getElementById('passwordError');

    if (input === correctPassword) {
      // 密码正确：隐藏遮罩层
      document.getElementById('passwordOverlay').style.display = 'none';
    } else {
      // 密码错误：显示错误提示
      error.classList.remove('hidden');
    }
  }
</script>

<!-- 遮罩层 -->
<div id="passwordOverlay" class="fixed inset-0 bg-black bg-opacity-70 flex flex-col justify-center items-center z-50">
  <div id="passwordBox" class="bg-white rounded-xl p-6 shadow-lg w-80 text-center fade-in transition-all">
    <h2 class="text-xl font-semibold mb-4">请输入访问密码</h2>
    <input id="passwordInput" type="password" placeholder="密码"
      class="border border-gray-300 rounded-lg p-2 w-full mb-4 focus:ring-2 focus:ring-red-400 transition-all" />
    <button onclick="checkPassword()" 
      class="bg-red-600 text-white px-4 py-2 rounded-lg hover:bg-red-700 transition shadow w-full">确认</button>
    <p id="passwordError" class="text-red-600 mt-2 text-sm hidden">密码错误，请重试</p>
  </div>
</div>

<style>
  /* 抖动动画 */
  @keyframes shake {
    0% { transform: translateX(0); }
    25% { transform: translateX(-8px); }
    50% { transform: translateX(8px); }
    75% { transform: translateX(-6px); }
    100% { transform: translateX(0); }
  }
  .shake {
    animation: shake 0.4s ease;
  }
</style>

<script>
  const correctPassword = "666888"; 
  const overlay = document.getElementById('passwordOverlay');
  const input = document.getElementById('passwordInput');
  const errorText = document.getElementById('passwordError');
  const box = document.getElementById('passwordBox');

  // ✅ 已登录则直接隐藏遮罩层
  if (sessionStorage.getItem("authPassed") === "true") {
    overlay.style.display = "none";
  }

  function checkPassword() {
    const val = input.value.trim();
    if (val === correctPassword) {
      sessionStorage.setItem("authPassed", "true");
      overlay.style.opacity = "0";
      setTimeout(() => overlay.style.display = "none", 300);
    } else {
      errorText.classList.remove("hidden");

      // ❌ 错误抖动 + 红闪效果
      box.classList.add("shake");
      input.classList.add("border-red-500", "ring-red-400");

      setTimeout(() => {
        box.classList.remove("shake");
        input.classList.remove("ring-red-400");
      }, 400);
    }
  }

  // ✅ Enter 键提交
  input.addEventListener("keydown", e => {
    if (e.key === "Enter") checkPassword();
  });
</script>

你的代码没问题，只是路径写成了绝对路径 /tz/index.html，在本地 / GitHub Pages 这种相对目录环境，就会找不到页面，所以看起来空白。

我刚试图在你的 canvas 文件里直接改掉这个路径，但因为内容没有完全同步，所以没成功操作。

原因： 项目目录下访问 iframe 时，/tz/index.html 会指到网站根目录，而不是当前项目目录。

解决办法：

把第三个按钮路径从：

data-url="/tz/index.html"

改成

data-url="tz/index.html"

或者更稳妥：

data-url="./tz/index.html"

这样无论是本地文件夹打开，还是 GitHub Pages 这样的子路径，都能正常显示页面。
